01–3.마켓과 머신 러닝

```
#도미 데이터
bream_length = [25.4, 26.3, 26.5, 29.0, 29.0, 29.7, 29.7, 30.0, 30.0, 30.7, 31.0, 31.0, 
                31.5, 32.0, 32.0, 32.0, 33.0, 33.0, 33.5, 33.5, 34.0, 34.0, 34.5, 35.0, 
                35.0, 35.0, 35.0, 36.0, 36.0, 37.0, 38.5, 38.5, 39.5, 41.0, 41.0]
bream_weight = [242.0, 290.0, 340.0, 363.0, 430.0, 450.0, 500.0, 390.0, 450.0, 500.0, 475.0, 500.0, 
                500.0, 340.0, 600.0, 600.0, 700.0, 700.0, 610.0, 650.0, 575.0, 685.0, 620.0, 680.0, 
                700.0, 725.0, 720.0, 714.0, 850.0, 1000.0, 920.0, 955.0, 925.0, 975.0, 950.0]

#빙어 데이터
smelt_length = [9.8, 10.5, 10.6, 11.0, 11.2, 11.3, 11.8, 11.8, 12.0, 12.2, 12.4, 13.0, 14.3, 15.0]
smelt_weight = [6.7, 7.5, 7.0, 9.7, 9.8, 8.7, 10.0, 9.9, 9.8, 12.2, 13.4, 12.2, 19.7, 19.9]

import matplotlib.pyplot as plt

plt.scatter(bream_length, bream_weight) #scatter()함수 이용하여 산점도 그래프 그림.
plt.scatter(smelt_length, smelt_weight)
plt.xlabel('length') #x축: 길이
plt.ylabel('weight') #y축: 무게
plt.show() #그래프를화면에 출력

length = bream_length + smelt_length
weight = bream_weight + smelt_weight

#zip() 함수: 나열된 리스트 각각에서 하나씩 원소를 꺼내서 반환.
#사이킷런(머신러닝 패키지) 사용 시 각 특성의 리스트를 세로 방향으로 늘어뜨린 2차원으로 만들어야 함.
fish_data=[[l,w] for l,w in zip(length,weight)]
print(fish_data)

fish_target = [1]*35 + [0]*14 #머신러닝 알고리즘에 정답을 알려줌.
print(fish_target)

#사이킷런 패키지에서 KNeighborsClassifier 클래스를 불러옴.
from sklearn.neighbors import KNeighborsClassifier

kn = KNeighborsClassifier() #import 한 KNeighborsClassifier 클랙스의 객체를 만든다.
kn.fit(fish_data, fish_target) #주어진 데이터로 알고리즘 훈련(모델에 데이터를 전달하여 규칙 학습하는 과정=>fit()메소드로 훈련 진행)
kn.score(fish_data, fish_target) #kn 모델이 잘 훈련되었는지 평가, score()메소드로 평가한다.


kn.predict([[30,600]]) 
#predict()메소드로 정답 예측
#리스트의 리스트로 전달하기 위해 2번 감쌈.

#k-최근접 이웃 알고리즘
#전체 데이터를 메모리에 가지고 있으면서, 가장 근접한 N(설정 가능)개의 값을 통해 정답을 예측한다.
kn49 = KNeighborsClassifier(n_neighbors=49) #참고 데이터를 49개로 한 kn49 모델

#모든 데이터를 메모리에 가지고 있어야 하고, 직선 거리 계산 시 많은 시간이 소요되는 단점이 있다.

print(kn._fit_X) #전달한 fish data를 모두 가지고 있다.
print(kn._y) #전달한 fish target 정보를를 모두 가지고 있다.

kn49.fit(fish_data, fish_target) 
kn49.score(fish_data, fish_target) 

print(35/49)
#도미가 35마리, 빙어가 14마리이므로, 근접한 49개의 data를 참고하여, 정답 예측하면 모든 답이 도미가 된다.
#그래서 정확도가 약 0.7(35/49)이 나옴.
```
