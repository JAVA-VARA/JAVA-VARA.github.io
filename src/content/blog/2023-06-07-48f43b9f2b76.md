```
import java.util.*;

public class Solution {
    public int[] solution(int []arr) {
                ArrayList<Integer> arr2 = new ArrayList<Integer>();

        arr2.add(arr[0]);

        for (int i = 1; i < arr.length; i++) {
            if (arr[i] == arr[i - 1])
                continue;
            else {
                arr2.add(arr[i]);
            }
        }
        int[] answer = new int[arr2.size()];
        answer = arr2.stream()
                .mapToInt(i -> i)
                .toArray();

        return answer;
    }
}
```
