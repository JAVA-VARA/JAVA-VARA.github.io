### Solution_1

```
package Programmers_Solution;

import java.util.*;

public class GreedyCloth {
    public int solution(int n, int[] lost, int[] reserve) {
        int answer = 0;

        //currentStatus setting
        HashMap<Integer, Integer> currStatus = new HashMap<>();
        for(int i=1; i <= n ; i++ ){
            currStatus.put(i , 1);
        }
        for(int i : lost){
            currStatus.put(i, currStatus.get(i) -1);
        }
        for(int i : reserve){
            currStatus.put(i, currStatus.get(i) +1);
        }
        
        for(int key: currStatus.keySet()){
            if(currStatus.get(key) == 0){
                if(currStatus.containsKey(key-1) && currStatus.get(key-1) ==2){
                    currStatus.put(key, currStatus.get(key) + 1);
                    currStatus.put(key-1, currStatus.get(key-1) -1);
                } else if (currStatus.containsKey(key+1) && currStatus.get(key+1) ==2) {
                    currStatus.put(key, currStatus.get(key) + 1);
                    currStatus.put(key+1, currStatus.get(key+1) -1);
                }
            }
        }

        //최종적으로 체육복을 한벌이상 가지고 있는 학생
        for(int key: currStatus.keySet()){
            if(currStatus.get(key) > 0){
                answer += 1;
            }
        }
        return answer;
    }

    public static void main(String[] args) {
        GreedyCloth greedyCloth = new GreedyCloth();
        System.out.println(greedyCloth.solution(5, new int[] {2,4}, new int[] {1,3,5}));
        System.out.println(greedyCloth.solution(5, new int[] {2,4}, new int[] {3}));
        System.out.println(greedyCloth.solution(3, new int[] {3}, new int[] {1}));
        System.out.println(greedyCloth.solution(7, new int[] {1,2,3}, new int[] {1,2,3}));
        System.out.println(greedyCloth.solution(7, new int[] {1,3}, new int[] {1,3,5}));
    }
}
```
