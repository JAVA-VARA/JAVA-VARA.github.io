## 07–2. 이벤트 활용(2)

[8]체크 박스 활용하기

```
<!DOCTYPE html>
<html lang="en">
<head>
    <title>Document</title>

    
    <script>
    //체크박스 활용하기: 체크 상태를 확인할 때 "checked 속성" 사용

    /*
    1. 체크 상태 일때만 타이머 증가
    2. change 이벤트가 발생했을 때 체크박스의 체크 상태 확인
    3. setInterval() 함수 또는 clearInterval() 함수 실행
    */

    document.addEventListener('DOMContentLoaded',()=>{
        let [timer,timerId] = [0,0] 

        const h1 = document.querySelector('h1')
        const checkbox = document.querySelector('input')

        checkbox.addEventListener('change', (event)=>{
            if(event.currentTarget.checked){
            //체크상태
            timerId = setInterval(()=>{
                timer += 1
                h1.textContent = `${timer}초`
            }, 10)
            }else{
            //체크상태해지
            clearInterval(timerId)
            } 
        })
     })
    </script>
</head>
<body>
    
        <input type = "checkbox">
        <span>타이머활성화</span>
        <h1></h1>

    
    
</body>
</html>
```
