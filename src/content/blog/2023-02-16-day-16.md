## DAY 16. 거북이 대포 만들기

각도를 맞춰 대포를 발사하여 목표 지점을 맞히는 게임

[명세]  
#1. 키보드 방향키로 발사 각도 조절(turn_up/turn_down 함수 이용)

#2. 스페이스바로 대포 발사(fire함수)

#3. 화살촉 모양의 대포가 하늘로 날아감

#4. 목표 지점을 맞히면 “good”, 빗나가면 “bad”  
(발사 각 조정 가능, 발사하는 힘 조절 x)

```
import turtle as t
import random

def turn_up():
    t.left(3)

def turn_down():
    t.right(3)

def fire():
    ang = t.heading() #현재 거북이가 바라보는 각도 기억
    while t.ycor()>0: #거북이가 땅 위에 있는 동안 반복
        t.forward(15) #15만큼 앞으로 이동
        t.right(5) #오른쪽으로 1도 회전

        #while 반복분을 빠져나오면 거북이가 땅에 닿은 상태입니다.

    d = t.distance(target,0)
    t.sety(random.randint(10,100))
    if d<25: #거리 차이가 25보다 작으면 목표 지점에 명중한 것으로 처리
        t.color("blue")
        t.write("Good", False, "center",("",10))
    else:
        t.color("red")
        t.write("Fail", False,"center",("",10))
    t.color("yellow") #거북이 색을 검은색으로 되돌림.
    t.goto(-200,10) #처음 발사했던 곳으로 되돌린다.
    t.setheading(ang) #처음 기억해 둔 각도로 되돌린다.

#땅을 그립니다.
t.goto(-300,0)
t.down()
t.goto(300,0)

#목표 지점을 설정하고 그립니다.
target = random.randint(50, 150) #목표 지점을 50~150사이 임의의 숫자로 지정
t.pensize(10)
t.color("green")
t.up()
t.goto(target-25,2)
t.down()
t.goto(target+25,2)

#거북이 색을 검정색으로 지정하고 처음 발사할 곳으로 되돌린다.
t.color("black")
t.up()
t.goto(-200,10)
t.setheading(20)

#거북이가 동작하는데 필요한 설정 지정
t.onkeypress(turn_up,"Up")
t.onkeypress(turn_down,"Down")
t.onkeypress(fire,"space")


t.listen()
```

## DAY17,18 터틀런 만들기

```
#터틀런 만들기

import turtle as t
import random

score = 0
playing = False

te=t.Turtle()
te.shape("turtle")
te.color("red")
te.speed(0)
te.up()
te.goto(0,200)

ts = t.Turtle()
ts.shape("circle")
ts.color("green")
ts.up()
ts.goto(0,-200)

def turn_right():
    t.setheading(0)

def turn_up():
    t.setheading(90)
def turn_left():
    t.setheading(180)
def turn_down():
    t.setheading(270)

def start():
    global playing
    if playing == False:
        #playing 값이 False라면 아직 게임 시작 전이므로,
        #게임 시작을 위해 playing 값을 true로 변경, play()함수 실행 및 메시지 삭제
        playing =  True
        t.clear()
        play()

def play():
    global score
    global playing
    t.forward(10)
    if random.randint(1,5) == 3: #20%확률로 악당 거북이가 따라옴.
        ang = te.towards(t.pos())
        te.setheading(ang)
    speed = score+5 #score가 올라갈때마다 악당 거북이 스피드 빨라진다.

    if speed > 15:
        speed = 15
    te.forward(speed)

    if t.distance(te)<12:

        text = "Score"+str(score)
        message("Game Over", text)
        playing = False
        score = 0

    if t.distance(ts)<12:
        score = score + 1 #초록 공을 먹을 때마다 score 1씩 증가
        t.write(score) #점수를 화면에 표시한다.
        star_x=random.randint(-230,230)
        star_y=random.randint(-230,230)
        ts.goto(star_x,star_y)

    if playing: #playing 변수 값이 true라면 0.1초마다 play 함수 실행
        t.ontimer(play,100)

def message(m1,m2):
    t.clear()
    t.goto(0,100)
    t.write(m1, False, "center",("",20))
    t.goto(0,-100)
    t.write(m2,False,"center",("",20))
    t.home()

t.title("Turtle Run")
t.setup(500,500)
t.bgcolor("orange")
t.shape("turtle")
t.speed(0)
t.up()
t.color("white")

t.onkeypress(turn_right,"Right")
t.onkeypress(turn_up,"Up")
t.onkeypress(turn_left,"Left")
t.onkeypress(turn_down,"Down")
t.onkeypress(start,"space")

t.listen()

message("Turtle Turtle Run! Run!","[press the Space to start]")
```
