### Flask 기본 기능

```
from flask import Flask, g, redirect, render_template, request, url_for, flash

myapp = Flask(__name__)


@myapp.route("/")
def index():
    return "hello flask"

@myapp.route(
    "/hello/<name>",
    methods=["GET", "POST"],
    endpoint="hello-endpoint",  # HTTP 메소드는 GET으로 지정
)  # hello-endpoint 라는 이름의 end-point 만들기
def hello(name):
    return f"hollo {name}!!!"


@myapp.route("/name/<name>")  # end-point 작성
def show_name(name):
    return render_template("index.html", name=name)


"""
@myapp.route 라는 데코레이터 Rule에 변수를 지정할 수 있다.
"""

# 라우팅 정보 출력하기
with myapp.test_request_context():  
# 플라스크의 테스트용 함수. 실제로 요청 없이 앱의 동작을 확인하기 위해 사용한다.
    print(url_for("index"))
    print(url_for("hello-endpoint", name="World"))
    print(url_for("show_name", name="AK", page="1"))

# 애플리케이션 컨텍스트를 취득하여 stack에 push
ctx = myapp.app_context()
ctx.push()
# # current_app에 접근할 수 있게된다.
# print(current_app.name)

# # 전역 임시 영역에 값을 설정한다.
g.connection = "connection"
print(g.connection)

with myapp.test_request_context("/users?updated = true"):
    print(request.args.get("updated"))
```

1. myapp.py

```
# 문의 폼을 반환하는 contact의 엔드포인트를 만든다.


@myapp.route("/contact")
def contact():
    return render_template("contact.html")


# 문의 폼 처리, 문의 완료 화면 반환하는 contact_complete의 엔드포인트를 만든다.
@myapp.route(
    "/contact/complete", methods=["GET", "POST"]
)  # 문의 폼을 불러오는 GET, 문의 폼을 보내는 POST 메소드를 허가한다.
def contact_complete():
    if request.method == "POST":
        # form의 속성을 사용해서 폼의 값을 얻는다.
        # username = request.form.get("username")
        username = request.form["username"]
        email = request.form["email"]
        description = request.form["description"]

        # 입력체크
        is_valid = True

        if not username:
            flash("사용자명은 필수입니다.")
            is_valid = False

        if not email:
            flash("이메일은 필수입니다.")
            is_valid = False

        try:
            validate_email(email)
        except EmailNotValidError:
            flash("메일 주소의 형식로 입력해 주세요.")
            is_valid = False

        if not description:
            flash("문의 내용은 필수입니다.")
            is_valid = False

        if not is_valid:
            return redirect(url_for("contact"))

        # 이메일을 보낸다.(TBC)

        flash("문의해주셔서 감사합니다. 이제 그만 가보셔도 됩니다.")

        # CONTACT 엔드포인트로 리다이렉트한다.
        return redirect(url_for("contact_complete"))  
        # POST의 경우 문의 완료 엔드포인트로 리다이렉트한다.
    return render_template("contact_complete.html")  
    # GET 의 경우 문의 완료 화면을 반환한다.
```

## ERROR

- ***jinja2.exceptions.templatesyntaxerror: tag name expected   
 →jinja2에서 for문이나 if 문을 사용할 때 *{% %}에 넣어서 사용해야하나, 오타로 인해 발생한 error 였다.

- ***400 Bad request. The browser (or proxy) sent a request that this server could not understand.  
 →***서버의 flask에서 받아들여야 하는 POST parameter가 없을 때 발생하는 error. flask에서 request.form[‘파라미터_이름’]으로 받아들이는 목록이 일치하는지 확인해야 한다.

이외에도 많은 error 가 발생했는데 대부분의 경우가 오타로 인한 error 였다. 오타만 줄여도 코딩에 들어가는 시간을 많이 단축 시킬 수 있으므로, 더 주의를 기울여야겠다.
