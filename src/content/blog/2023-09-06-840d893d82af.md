[**코딩테스트 연습 - H-Index**  
*H-Index는 과학자의 생산성과 영향력을 나타내는 지표입니다. 어느 과학자의 H-Index를 나타내는 값인 h를 구하려고 합니다. 위키백과에 따르면, H-Index는 다음과 같이 구합니다. 어떤 과학자가 발표한…*school.programmers.co.kr](https://school.programmers.co.kr/learn/courses/30/lessons/42747)[](https://school.programmers.co.kr/learn/courses/30/lessons/42747)

```
package Programmers_Solution.LV02;

import java.util.Arrays;
import java.util.Collections;
import java.util.LinkedList;

public class Hindex {
    public int solution(int[] citations) {
        Integer[] citationsReverse = Arrays.stream(citations).boxed().toArray(Integer[]::new);
        Arrays.sort(citationsReverse, Collections.reverseOrder());
        LinkedList<Integer> list = new LinkedList<>();

        int answer = 0;
        for(int i = 0; i<citationsReverse.length; i++){
            list.add(citationsReverse[i]);
            if(citationsReverse[i] < list.size()){
                break;
            }
            answer += 1;
        }
        return answer;
    }
    public static void main(String[] args) {
        Hindex hindex = new Hindex();
        System.out.println(hindex.solution(new int[] {3, 0, 6, 1, 5} )); //3
        System.out.println(hindex.solution(new int[] {0,5,6,7,8} )); //4
        System.out.println(hindex.solution(new int[] {9,9,9,9,1} )); //4
    }
}
```
