```
package Programmers_Solution;

import java.util.ArrayList;
import java.util.List;

public class MakeSosu {
    public int solution(int[] nums) {

        List<Integer> primeNumbers = new ArrayList<>();
        int sumThree = 0;
        int count = 0;
        int answer = 0;

        for (int i = 0; i < nums.length - 2; i++) {
            for (int j = i + 1; j < nums.length - 1; j++) {
                for (int k = j + 1; k < nums.length; k++) {
                    if (nums[i] != nums[j] && nums[i] != nums[k] && nums[j] != nums[k]) {
                        sumThree = nums[i] + nums[j] + nums[k];
                    }
                    if (sumThree % 2 != 0) {
                        primeNumbers.add(sumThree);
                    }
                }
            }
        }
        for (int primeNumber:primeNumbers) {
            for (int i = 2; i*i <= primeNumber; i ++ ) {
                if (primeNumber % i == 0) {
                    count += 1;
                }
                if (primeNumber % i != 0) {
                    continue;
                }
            }if(count == 0){
                answer += 1;

            }count = 0;
        }
        return answer;
    }

    public static void main(String[] args) {
        MakeSosu ms = new MakeSosu();
        System.out.println(ms.solution(new int[]{1, 2, 3, 4})); //1
        System.out.println(ms.solution(new int[]{1, 2, 7, 6, 4})); //4
    }
}
```

### [Sol2]

```
package Programmers_Solution;

import java.util.ArrayList;
import java.util.List;

public class MakeSosu {
    public int solution(int[] nums) {

        List<Integer> primeNumbers = new ArrayList<>();
        for (int i = 0; i < nums.length - 2; i++) {
            for (int j = i + 1; j < nums.length - 1; j++) {
                for (int k = j + 1; k < nums.length; k++) {
                    if (nums[i] != nums[j] && nums[i] != nums[k] && nums[j] != nums[k]) {
                        // 변수를 if 문 안에 넣어주었다.
                        int sumThree = nums[i] + nums[j] + nums[k];
                        primeNumbers.add(sumThree);
                    }
                }
            }
        }
        //변수 선언 위치 변경
        int count = 0;
        int answer = 0;
        for (int primeNumber:primeNumbers) {
            //Math 메서드를 사용하여 비교문 변경하였다.
            for (int i = 2; i <= (int)Math.sqrt(primeNumber); i++ ) {
                if (primeNumber % i == 0) count += 1;
            }
            if(count == 0) answer += 1;
            count = 0;
        }
        return answer;
    }
}
```
