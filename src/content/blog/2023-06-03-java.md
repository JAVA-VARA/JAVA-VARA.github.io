import java.util.Arrays;

public class Divisible2 {
    public int[] divisible(int[] array, int divisor) {

        int retArraySize = 0;
        for (int i = 0; i < array.length; i++) {
            if (array[i] % divisor == 0) {
                retArraySize += 1;
            }
        }
        int[] ret = new int[retArraySize];
        int j = 0;
        for (int i = 0; i < array.length; i++) {
            if (array[i] % divisor == 0) {
                ret[j++] = array[i];
            }
        }
        return ret;

    }

    public static void main(String[] args) {
        Divisible2 div = new Divisible2();
        int[] array = { 1, 2, 3, 4, 5 };
        System.out.println(Arrays.toString(div.divisible(array, 5)));
    }

}

## String을 숫자로 바꾸기(Sol 1)

```
public class StrToInt {
    public int getStrToInt(String str) {
        // 결과값이 int형이므로 int형을 담을 수 있는 int형 result 변수 선언
        int result = 0;

        /*
         * +,- , 아무것도 가지지 않는 것 3가지 경우의 수가 있음.
         * 부호를 나눌 int형 변수 sign 선언
         * 양수인 경우: *1, 음수인 경우: *(-1),
         * 
         */
        int sign = 1;

        // 첫번째 문자부터 숫자가 들어오는 경우, index를 0으로 초기화
        // 그 외의 경우 index가 1부터 시작하게 함.
        int index = 1;

        // 문자열로 들어온 값을 정수로 변경

        // 첫번째 문자를 얻는 것은 chartAt()메서드 사용.
        char ch = str.charAt(0);

        // 첫번째 문자를 통해 양수 or 음수, 부호 유무 판단
        if (ch == '+') {
            sign = 1;
        } else if (ch == '-') {
            sign = -1;
        } else {
            sign = 1;
            // 앞에 부호가 없으니깐 index 0부터 숫자 시작하므로 0으로 초기화.
            index = 0;
        }

        for (int i = index; i < str.length(); i++) {
            result = result * 10 + (str.charAt(i) - '0');
        }

        return sign * result;
    }

    public static void main(String[] args) {
        StrToInt convert = new StrToInt();
        System.out.println(convert.getStrToInt("-1234"));
        System.out.println(convert.getStrToInt("+1234"));
        System.out.println(convert.getStrToInt("1234"));

    }
}
```

public class ReverseStr {
    public String reverseStr(String str) {
        
        //문자열을 char형 배열로 변환한다.
        char[] chars = str.toCharArray();
        char temp;

        for (int i = 0; i < chars.length; i++) {
            for (int j = 0; j < chars.length; j++) {
                //배열의 원소들을 꺼내어 비교한다.
                if (chars[i] < chars[j]) {

                    temp = chars[i];
                    chars[i] = chars[j];
                    chars[j] = temp;
                }

            }
        }
        return new String(chars);
    }

    public static void main(String[] args) {
        ReverseStr rs = new ReverseStr();
        System.out.println(rs.reverseStr("Zbcdefg"));
    }
}

public class ReverseInt {
    public int reverseInt(int n) {
        // 1. 정수 자릿수 구하기
        int i = 1;
        int count = 0;
        while (i <= n) {
            i = i * 10;
            count++;
        }
        int arrCount = 0;
        i = i / 10;
        // 자릿수 크기만큼 정수 배열 선언
        int[] arr = new int[count];

        // 배열에 각 자리 수 저장.
        while (n > 0) {
            arr[arrCount] = n / i;
            n = n % i;
            i = i / 10;
            arrCount++;
        }
        // 내림차순으로 배열 정렬
        int temp = -1;
        for (int i2 = 0; i2 < arr.length; i2++) {
            for (int j = i2 + 1; j < arr.length; j++) {
                if (arr[i2] < arr[j]) {
                    temp = arr[i2];
                    arr[i2] = arr[j];
                    arr[j] = temp;
                }
            }
        }
        // 베열에 있는 값을 이용해 다시 정수를 만든다.
        int value = 0;
        if (arr.length > 0) {
            value = arr[0];
            if (arr.length > 1) {
                for (int i3 = 1; i3 < arr.length; i3++) {
                    value = value * 10 + arr[i3];
                }
            }
        }

        return value;
    }

    public static void main(String[] args) {
        ReverseInt ri = new ReverseInt();
        System.out.println(ri.reverseInt(118372));
    }
}

## 최댓값과 최솟값

    /*
     * 1. split()을 이용하여 공백을 기준으로 문자열을 자른다.=>잘린 문자는 배열로 반환된다.
     * 2. 최댓값과 최솟값을 담을 변수 선언 and 변수에 배열의 첫번째 요소를 담는다.
     * 3. 문자열을 정수형 배열로 바꿔준다. by Integer.parseInt()
     * 4. 두 번째 요소부터 비교 시작
     * 5. 최댓값 변수 / 최솟값 변수 경우에 따라 변경
     */

public class GetMinMaxString {
    public String getMinMaxString(String str) {
        String[] strArr = str.split(" ");
        int maxValue = Integer.parseInt(strArr[0]);
        int minValue = Integer.parseInt(strArr[0]);

        for (int i = 1; i < strArr.length; i++) {
            int value = Integer.parseInt(strArr[i]);

            if (value > maxValue) {
                maxValue = value;
            }
            if (value < minValue) {
                minValue = value;
            }
        }
        return minValue + " " + maxValue;
    }

    public static void main(String[] args) {
        String str1 = "1 2 3 4";
        String str2 = "-1 -2 -3 -4";
        GetMinMaxString gmm = new GetMinMaxString();
        System.out.println(gmm.getMinMaxString(str1));
        System.out.println(gmm.getMinMaxString(str2));
    }
}

public class HarshadNumber {
    public boolean isHashard(int num) {
        int nsum = 0;
        int n = num;
        // boolean isHashard = false;

        while (n > 0) {
            nsum += n % 10;
            n = n / 10;
        }
        if (num % nsum == 0) {
            return true;
        } else {
            return false;
        }
    }

    public static void main(String[] args) {
        HarshadNumber hn = new HarshadNumber();
        System.out.println(hn.isHashard(10));
        System.out.println(hn.isHashard(18));

        System.out.println(hn.isHashard(13));
    }
}
