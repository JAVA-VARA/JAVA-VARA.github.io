### Solution_1

정답이지만, 시간 초과로 실패.

```
package Programmers_Solution;

import java.util.ArrayList;
import java.util.HashMap;

public class Malaton {
    public String solution(String[] participant, String[] completion) {
        String answer = "";
        HashMap<Integer, ArrayList<String>> totalCompletedPlayers = new HashMap<>(20);

        for(String completedPlayer : completion) {
            int lengthOfName = completedPlayer.length();
            totalCompletedPlayers.putIfAbsent(lengthOfName, new ArrayList<>());

            ArrayList<String> completedPlayerList = totalCompletedPlayers.get(lengthOfName);
            completedPlayerList.add(completedPlayer);

            totalCompletedPlayers.put(lengthOfName, completedPlayerList);
        }

        for(String attendee: participant) {
            int lengthOfName = attendee.length();

            if (!totalCompletedPlayers.containsKey(lengthOfName)) {
                answer = attendee;
                break;}

                if (totalCompletedPlayers.get(lengthOfName).contains(attendee)) {
                    ArrayList<String> completedPlayerList = totalCompletedPlayers.get(lengthOfName);
                    completedPlayerList.remove(attendee);
                    totalCompletedPlayers.put(lengthOfName, completedPlayerList);
                }
                else if (!totalCompletedPlayers.get(lengthOfName).contains(attendee)) {
                    answer = attendee;
                }
        }
        return answer;
    }
    public static void main(String[] args) {
        Malaton malaton = new Malaton();
        System.out.println(malaton.solution(
                new String[]{"leo", "kiki", "eden"},
                new String[]{"eden", "kiki"}));
        //"leo"

        System.out.println(malaton.solution(
                new String[]{"marina", "josipa", "nikola", "vinko", "filipa"},
                new String[]{"josipa", "filipa", "marina", "nikola"}));
        //vinko
        System.out.println(malaton.solution(
                new String[]{"mislav", "stanko", "mislav", "ana"},
                new String[]{"stanko", "ana", "mislav"}));
        //mislav
    }

}
```

### Solution_3

1. String 배열 participant와 completion을 각각 HashMap에 넣는다.
2. key값을 통해 value의 list에 접근하여, size를 비교
3. size가 다른 경우 break
4. 해당 value의 배열 에서 중복되는 내용 삭제 후 남은 것을 반환

```
package Programmers_Solution;

import java.util.ArrayList;
import java.util.Arrays;
import java.util.HashMap;
import java.util.LinkedList;

public class Maratons {
    public String solution(String[] participant, String[] completion) {
        String answer = "";
        // 상수 시간
        HashMap<Integer, LinkedList<String>> totalCompletedPlayers = new HashMap<>(20);
        HashMap<Integer, LinkedList<String>> totalParticipants = new HashMap<>(20);

        for(int i =1; i <= 20 ; i++){ // 20회 반복 (상수 회수)
            totalCompletedPlayers.put(i, new LinkedList<>());
            totalParticipants.put(i, new LinkedList<>());
        }
        
        Arrays.stream(completion).forEach(completedPlayer -> { //최대: O(n)
            int lengthOfName = completedPlayer.length(); // 상수 시간
            LinkedList<String> completedPlayerList = totalCompletedPlayers.get(lengthOfName); // 상수 시간
            completedPlayerList.add(completedPlayer); // 상수 시간
            totalCompletedPlayers.put(lengthOfName, completedPlayerList); // 상수 시간
        });

        Arrays.stream(participant).forEach(attendee -> { //최대: O(n)
            int lengthOfName = attendee.length(); // 상수 시간
            LinkedList<String> attendeeList = totalParticipants.get(lengthOfName); // 상수 시간
            attendeeList.add(attendee);// 상수 시간
            totalParticipants.put(lengthOfName, attendeeList);// 상수 시간
        });

        for(int i =1; i <= 20 ; i++){ // 20회 반복 (상수 회수)
            //get 메서드를 통해 배열을 순회하지 않고 key 값을 통해 value에 접근
            int numberOfParticipant = (totalParticipants.get(i)).size(); // 상수 시간
            int numberOfCompletion = (totalCompletedPlayers.get(i)).size(); // 상수 시간

            if(numberOfParticipant !=  numberOfCompletion ){
                LinkedList<String> a = totalParticipants.get(i);
                LinkedList<String> b = totalCompletedPlayers.get(i);

                if(!a.removeAll(b)){ // 최대: O(n)
                    a.removeAll(b); // 최대: O(n)
                    answer = a.getFirst();
                    // => 여기서 최악의 경우 O(n^2) 의 알고리즘 복잡도를 보일 것으로 예상
                }else{
                    HashMap<String, Integer> nameFrequency  = new HashMap<>();
                    for(String name: b){ // 최대: O(n)
                        nameFrequency.put(name, nameFrequency.getOrDefault(name, 0) + 1);
                    }
                    for(String name: a){ // 최대: O(n)
                        nameFrequency.put(name, nameFrequency.get(name) - 1);
                    }
                    for (String name : nameFrequency.keySet()) { // 최대: O(n)
                        if (nameFrequency.get(name) > 0) {
                            answer = name;
                            break;
                        }
                    }
                }
            }
        }
        return answer;
    }
}
```

위와 같은 과정을 거치면 대규모 연산을 2번만 시행하면 되기에 통과하지 않을까 했지만 효율성 test에서 탈락했다.

최적화를 위해 전체적인 코드를 보며 알고리즘 복잡도를 라인별로 확인해 보았고(자세한 사항은 주석 참조). *if 구문에서 O(n²)의 알고리즘 복잡도를 가지는 것으로 확인하였다.

**Solution_1**에서 시간 초과로 효율성 test에서 탈락했을 때 코드에서 개선 해야할 점은 알고리즘 복잡도였다.

그러나 Root cause에 대한 고민과 어느 부분에서 알고리즘 복잡도가 증가했는지 정확히 파악하지 않고 *“코드를 좀 더 효율적으로 작성해야겠다”* 라는 추상적인 생각 만을 가지고 단순히 로직만 변경하여 코드를 작성하니 문제를 해결할 수 없었다.

결과적으로 문제를 해결하기 까지 상당히 오랜 시간을 낭비하게 되었다. 처음부터 개선이 필요한 부분에 대해 고심한 후 정확한 목적성을 가지고 코드를 작성하였더라면 하는 아쉬움이 남는다.

### Solution_4

아래와 같은 방법으로 hashMap에 key, value를 추가하는 것으로 전체적인 코드를 변경하였다.

- key: 참가자 이름
- value: 1(default 값)
- key값이 동일한 경우(동명이인인 경우) 현재 value 값 +1

더불어 for문이나, if 구문 하단에 속한 코드는 전부 상수 시간 내에 연산이 가능한, 즉 participant나 completion 배열을 순회하지 않는 방향으로 코드를 수정함으로써 코드를 최적화 하였다.

```
package Programmers_Solution;
import java.util.HashMap;

public class MaratonNew {
    public String solution(String[] participant, String[] completion) {
        String answer = "";
        HashMap<String, Integer> failure = new HashMap<>();

        for(String s:participant){
            if(!failure.isEmpty() && failure.containsKey(s)){ // 최대: O(n) - 해시맵 조회
                failure.put(s,failure.get(s)+1); // 상수 시간
            }else{
                failure.put(s,1); // 상수 시간
            }
        }
        for(String s: completion){
            if(failure.containsKey(s)){ // 최대: O(n) - 해시맵 조회
                failure.put(s,failure.get(s)-1); // 상수 시간
            }
        }
        for(String key : failure.keySet()){ // 최대: O(n) - 해시맵 조회
            if(failure.get(key) == 1){ // 상수 시간
                answer = key; // 상수 시간
            }
        }
        return answer;
    }
}
(=> 최대 O(n) 의 시간이 소요되어, 전체 알고리즘 복잡도가 O(n)이다.)
```
