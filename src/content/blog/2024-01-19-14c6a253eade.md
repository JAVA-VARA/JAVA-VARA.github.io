[**코딩테스트 연습 - 가장 많이 받은 선물**  
*선물을 직접 전하기 힘들 때 카카오톡 선물하기 기능을 이용해 축하 선물을 보낼 수 있습니다. 당신의 친구들이 이번 달까지 선물을 주고받은 기록을 바탕으로 다음 달에 누가 선물을 많이 받을지 예측하려고 합니다. 두…*school.programmers.co.kr](https://school.programmers.co.kr/learn/courses/30/lessons/258712?language=java)[](https://school.programmers.co.kr/learn/courses/30/lessons/258712?language=java)

import java.util.*;

public class Present {
    public int solution(String[] friends, String[] gifts) {
        //선물 주고 받은 횟수 기록할 이중 배열 생성
        Map<String, Integer> map = new HashMap<>();
        int i = 0;
        for(String friend : friends){
            map.put(friend, i);
            i++;
        }

        int[][] giveAndTake = new int[friends.length][friends.length];
        for(String giver : friends){
            for(String taker: friends){
                giveAndTake[map.get(giver)][map.get(taker)] = 0;
            }
        }

        //선물 주고 받을 때 마다 +1 씩 증가하도록 로직 구성
        //#1 선물 주고 받은 현황
        String[][] giftsArray = new String[gifts.length][];
        int j =0;
        for(String s : gifts){
            String[] newGifts = s.split("\\s");
            giftsArray[j] = newGifts;
            j++;
        }

        //#2 선물 주고 받을 때 마다 +1 씩 증가하도록 로직 구성
        for(String[] s: giftsArray){
            String giver = s[0];
            String taker = s[1];
            giveAndTake[map.get(giver)][map.get(taker)] += 1;
        }

        //준 선물 갯수
        Map<String, List<String>> gt = new HashMap<>();
        for(String[] s : giftsArray){
            gt.putIfAbsent(s[0], new ArrayList<>());
        }

        for(String[] s : giftsArray){
            List<String> input = gt.get(s[0]);
            input.add(s[1]);
            gt.put(s[0], input);
        }

        Map<String, Integer> giftIdx = new HashMap<>();
        for(String s : friends){
            if(gt.get(s) != null){
                giftIdx.put(s, gt.get(s).size());
            }else {
                giftIdx.put(s, 0);
            }
        }

        Map<Integer, String> map2 = new HashMap<>();
        int q = 0;
        for(String friend : friends){
            map2.put(q, friend);
            q++;
        }

        //받은 선물 갯수
        int take = 0;
        for(int x = 0; x< giveAndTake.length; x++){
            for(int y = 0; y<giveAndTake.length; y++){
                take += giveAndTake[y][x];
            }
            giftIdx.put(map2.get(x), giftIdx.get(map2.get(x)) - take);
            take = 0;
        }

        //1. 배열 비교 2. 선물 지수 비교
        for (String key : map.keySet()) {
            map.put(key, 0);
        }

        for(int x = 0; x < giveAndTake.length; x++){
            for(int y = 0; y < giveAndTake.length; y++ ){

                if(x==y){
                    continue;
                }

                int XtoY = giveAndTake[x][y];
                int YtoX = giveAndTake[y][x];
                String friend1 = map2.get(x);
                String friend2 = map2.get(y);


                //선물 더 많이 준사람
                if(XtoY > YtoX){
                    String giveMore = map2.get(x);
                    int getGiftForNextMonth = map.get(giveMore);
                    map.put(giveMore, getGiftForNextMonth+1);
                }

                //선물지수가 더 높은 사람
                if(XtoY == YtoX) {
                    int giftIdx1 = giftIdx.get(friend1);
                    int giftIdx2 = giftIdx.get(friend2);


                    if(giftIdx1 > giftIdx2){
                        int getGiftForNextMonth = map.get(friend1);
                        map.put(friend1, getGiftForNextMonth+1);
                    }
                }
            }
        }

        int maxGift = 0;
        for (Map.Entry<String, Integer> entry : map.entrySet()) {
            if (entry.getValue() > maxGift) {
                maxGift = entry.getValue();
            }
        }

        return maxGift;
    }

    public static void main(String[] args) {
        Present present = new Present();
        System.out.println(present.solution(
                new String[] {"muzi", "ryan", "frodo", "neo"},
                new String[] {"muzi frodo", "muzi frodo", "ryan muzi", "ryan muzi", "ryan muzi", "frodo muzi", "frodo ryan", "neo muzi"}));

        System.out.println(present.solution(
                new String[] {"joy", "brad", "alessandro", "conan", "david"},
                new String[] {"alessandro brad", "alessandro joy", "alessandro conan", "david alessandro", "alessandro david"}));

        System.out.println(present.solution(
                new String[] {"a", "b", "c"},
                new String[] {"a b", "b a", "c a", "a c", "a c", "c a"}));

    }

}
