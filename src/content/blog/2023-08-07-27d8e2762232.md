### Solution_1

```
import java.util.Arrays;
import java.util.HashMap;
import java.util.LinkedList;

public class PersonalInfo_Expiry {
    public int[] solution(String today, String[] terms, String[] privacies){

        //1 terms ""기준 split 그리고 k,v 로 변환
        String[][] termsNew = new String[terms.length][];
        for(int i = 0; i < terms.length; i++){
            termsNew[i] = terms[i].split(" ");
        }
        HashMap<String, Integer> term = new HashMap<>();
        for (String[] strings : termsNew) {
            term.put(strings[0], Integer.parseInt(strings[1]) * 28);
        }
        //2 privacies split
        String[][] privaciesNew = new String[privacies.length][];
        for(int i = 0; i < privacies.length; i++){
            privaciesNew[i] = privacies[i].split(" ");
        }

        //수집기간 날짜만 담은 배열
        String[][] getPrivacy = new String[privaciesNew.length][3];
        for(int i = 0 ; i < privaciesNew.length; i++){
            getPrivacy[i] = privaciesNew[i][0].split("\\.");
        }

        //수집기간 약관만 담은 배열
        int[] condition = new int[privaciesNew.length];
        for(int i = 0 ; i < privacies.length; i++ ){
            condition[i] = term.get(privaciesNew[i][1]);
        }

        //today split
        String[] todayNew = today.split("\\.");

        //수집일 계산
        int[] collected = new int[privacies.length];
        for(int i = 0 ; i < privacies.length; i++){
            int year = Integer.parseInt(todayNew[0]) - Integer.parseInt(getPrivacy[i][0]);
            int month = Integer.parseInt(todayNew[1]) - Integer.parseInt(getPrivacy[i][1]);
            int date = Integer.parseInt(todayNew[2]) - Integer.parseInt(getPrivacy[i][2]);
            collected[i] = year*12*28 + month*28 + date;
        }

        //결과값 반환
        LinkedList<Integer> result = new LinkedList<>();
        for(int i = 0 ; i < collected.length ; i++){
            if(collected[i] - condition[i] >= 0){
                result.add(i+1);
            }
        }

        //list to array
        int[] answer = new int[result.size()];
        for (int i = 0; i < result.size(); i++) {
            answer[i] = result.get(i);
        }

        return answer;
    }

    public static void main(String[] args) {
        PersonalInfo_Expiry pie = new PersonalInfo_Expiry();
        for (int[] ints : Arrays.asList(

                pie.solution(
                "2022.05.19",
                new String[]{"A 6", "B 12", "C 3"},
                new String[]{"2021.05.02 A", "2021.07.01 B", "2022.02.19 C", "2022.02.20 C"}),

                pie.solution(
                "2020.01.01",
                new String[]{"Z 3", "D 5"},
                new String[]{"2019.01.01 D", "2019.11.15 Z", "2019.08.02 D", "2019.07.01 D", "2018.12.28 Z"}))) {

            System.out.println(Arrays.toString(ints));
        }

    }
}
```

### Solution_2

#1_일부 중복되는 코드를 메소드로 정의하여 사용하였다.

```
import java.util.Arrays;
import java.util.HashMap;
import java.util.LinkedList;

public class PersonalInfoValidation {
    public int[] solution(String today, String[] terms, String[] privacies){
        // #1
        String[][] termsNew = splitStringByBlank(terms);
        HashMap<String, Integer> term = new HashMap<>();
        for (String[] strings : termsNew) {
            term.put(strings[0], Integer.parseInt(strings[1]) * 28);
        }
        // #1
        int currDate = convertDateToInteger(today);
        String[][] privaciesNew = splitStringByBlank(privacies);

        int[] condition = new int[privaciesNew.length];

        LinkedList<Integer> consentedPeriod = new LinkedList<>();
        for(String[] strings: privaciesNew){
            // #1
            int collectDate = convertDateToInteger(strings[0]);

            consentedPeriod.add(currDate - collectDate);

        }

        for(int i = 0 ; i < privacies.length; i++ ){
            condition[i] = term.get(privaciesNew[i][1]);
        }

        LinkedList<Integer> result = new LinkedList<>();
        for(int i = 0 ; i < consentedPeriod.size(); i++){
            if(consentedPeriod.get(i) >= condition[i]){
                result.add(i+1);
            }
        }

        return result
                .stream()
                .mapToInt(integer -> integer)
                .toArray();
    }
    // #1
    private int convertDateToInteger(String date){
        String[] str = date.split("\\.");
        int year = Integer.parseInt(str[0]);
        int month = Integer.parseInt(str[1]);
        int day = Integer.parseInt(str[2]);

        return year*12*28 + month*28 + day;
    }

    // #1
    private String [][] splitStringByBlank(String[] arr){
        String[][] reArr = new String[arr.length][];

        for(int i = 0; i < arr.length; i++){
            reArr[i] = arr[i].split(" ");
        }
        return reArr;
    }
}
```

### Solution_3

solution_2와 동일한 코드. for문을 stream API와 람다 함수를 사용하여 표현하였다.

```
import java.util.Arrays;
import java.util.HashMap;
import java.util.LinkedList;
import java.util.stream.Collectors;
import java.util.stream.IntStream;

public class PersonalInfoValidation {
    public int[] solution(String today, String[] terms, String[] privacies){

        String[][] termsNew = splitStringByBlank(terms);
        HashMap<String, Integer> term = Arrays.stream(termsNew)
                .collect(Collectors
                        .toMap(strings -> strings[0], strings -> Integer.parseInt(strings[1]) * 28, (a, b) -> b, HashMap::new));


        int currDate = convertDateToInteger(today);
        String[][] privaciesNew = splitStringByBlank(privacies);

        int[] condition = new int[privaciesNew.length];

        LinkedList<Integer> consentedPeriod = Arrays.stream(privaciesNew)
                .mapToInt(strings -> convertDateToInteger(strings[0]))
                .mapToObj(collectDate -> currDate - collectDate)
                .collect(Collectors.toCollection(LinkedList::new));

        IntStream.range(0, privacies.length).forEach(i -> condition[i] = term.get(privaciesNew[i][1]));

        LinkedList<Integer> result = IntStream.range(0, consentedPeriod.size())
                .filter(i -> consentedPeriod.get(i) >= condition[i]).mapToObj(i -> i + 1)
                .collect(Collectors.toCollection(LinkedList::new));

        return result
                .stream()
                .mapToInt(integer -> integer)
                .toArray();
    }

    private int convertDateToInteger(String date){
        String[] str = date.split("\\.");
        int year = Integer.parseInt(str[0]);
        int month = Integer.parseInt(str[1]);
        int day = Integer.parseInt(str[2]);

        return year*12*28 + month*28 + day;
    }

    private String [][] splitStringByBlank(String[] arr){

        return Arrays.stream(arr)
                .map(s -> s.split(" "))
                .toArray(String[][]::new);
    }
}
```
