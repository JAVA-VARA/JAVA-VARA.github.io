package Programmers_Solution.LV01;

import java.util.ArrayList;
import java.util.Arrays;

public class MockTTest {
    public int[] solution(int[] answers) {
       
        int[] studentNo1 = {1, 2, 3, 4, 5};
        int[] studentNo2 = {2, 1, 2, 3, 2, 4, 2, 5};
        int[] studentNo3 = {3, 3, 1, 1, 2, 2, 4, 4, 5, 5};

        int[] result = new int[3];
        result[0] = answerReader(studentNo1,answers);
        result[1] = answerReader(studentNo2,answers);
        result[2] = answerReader(studentNo3,answers);

        ArrayList<Integer> bestStudent = new ArrayList<>();
        int maxScore = Math.max(Math.max(result[0], result[1]), result[2]);

        for (int i = 0; i < result.length; i++) {
            if (result[i] == maxScore) {
                bestStudent.add(i);
            }
        }

        int[] answer = new int[bestStudent.size()];
        for (int i = 0; i < bestStudent.size(); i++) {
            answer[i] = bestStudent.get(i)+1;
        }
        
        return answer;
    }

    int answerReader(int[] studentAnswer, int[] correctAnswer) {
        int correct = 0;
        int pattern = studentAnswer.length;
        for (int i = 0; i < correctAnswer.length; i++) {
            if (correctAnswer[i] == studentAnswer[i % pattern]) {
                correct += 1;
            }
        }
        return correct;
    }

    public static void main(String[] args) {
        MockTTest mouiTest = new MockTTest();
        System.out.println(Arrays.toString(mouiTest.solution(new int[]{1, 2, 3, 4, 5})));
        System.out.println(Arrays.toString(mouiTest.solution(new int[]{1,3,2,4,2})));
    }

}
