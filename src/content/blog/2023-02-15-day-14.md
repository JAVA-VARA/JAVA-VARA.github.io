## DAY 14. 계산 맞히기 게임

[명세]

#1. random 모듈을 이용해 사칙연산 문제를 임의로 만들어냄.

#2. 사용자가 계산해서 답 입력

#3. 정담 여부 확인 후 점수를 매긴다.

#4. 5번 반복해서 전체 정답 수를 알려준다.

```
import random

def make_question():
    a = random.randint(1,40)
    b = random.randint(1,20)
    op = random.randint(1,3)

    q=str(a)

    if op == 1:
        q=q+"+"

    if op == 2:
        q=q+"-"
    if op == 3:
        q=q+"*"

    q = q+str(b)

    return q

sc1 = 0
sc2 = 0

for x in range(5):
    q=make_question()
    print(q)
    ans = input("=")
    r=int(ans)

    if eval(q) == r:
        print("정답")
        sc1 = sc1 + 1
    else:
        print("오답")
        sc2 = sc2 + 1

print("정답:", sc1, "오답", sc2)
if sc2 == 0:
    print("너는 천재 지니우스!")
```

## DAY 15. 타자 게임 만들기

- list: 여러 정보를 하나로 묶어서 저장하고 관리할 수 있는 기능
- random.choice: list 내 자료 중 임의로 하나 선택

[명세]

#1. 사용자가 Enter를 누를 때 까지 대기

#2. 사용자에게 임의로 문제를 보여주고 타자 입력 받아 처리.(5회 반복)

#3. 결과 계산해서 보여주기

```
import random
import time

#단어 리스트: 여기에 단어 추가하면 문제에 나온다.

w=["cat", "dog", "fox", "monkey", "mouse", "panda", "forg", "snake", "wolf"]

n=1

print("[타자게임] 준비되면 enter키를 눌러주세요")
input() #t사용자가 엔터 누를 때 까지 기다림.
start = time.time() #시작시간 기록

q=random.choice(w) #단어 리스트에서 임의로 하나 선택

while n <= 5:
    print("*#문제",n)
    print(q) #문제 보여줌
    x=input() #사용자 입력

    if q == x:
        print("통과")
        n = n+1 #정답을 입력한 경우 문제 번호 1 증가
        q = random.choice(w) #새로운 문제 제출
    else:
        print("오타! 재도전") #오타가 난 경우 재작성

end = time.time() #끝난 시간 기록
et = end - start #실제 걸린 시간 계산
et = format(et,".2f") #소수 둘째 자리까지 표기
print("타자시간:", et, "초")
```
