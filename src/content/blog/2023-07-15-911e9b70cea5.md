import java.util.ArrayList;
import java.util.List;

public class WeaponOfKnight {
    public int solution(int number, int limit, int power) {

        List<Integer> knightPower = new ArrayList<>();

        for(int i = 1; i <= number; i++){

            int count = 0;
            for(int j=1; j <= (int)Math.sqrt(i); j++) {
                if(j*j == i){count += 1;}
                else if (i % j == 0) {count += 2;}
            }

            if(count <= limit){knightPower.add(count);}
            else{knightPower.add(power);}
        }

        int answer = knightPower
                .stream()
                .mapToInt(Integer::intValue)
                .sum();

        return answer;
    }

    public static void main(String[] args) {
        WeaponOfKnight wok = new WeaponOfKnight();
        System.out.println(wok.solution(5,3,2)); //10
        System.out.println(wok.solution(10,3,2)); //21
    }
}

import java.util.PriorityQueue;

public class Painting {
    public int solution(int n, int m, int[] section) {
        
        //1. section[0] ~ section[0]+m 까지의 값을 q값에 넣어 준다.
        PriorityQueue<Integer> paintWall = new PriorityQueue<>();
        for(int i = section[0]; i < section[0]+m ; i++ ){
            if(i <= n){
                paintWall.add(i);
            }
        }

        //2. 우선순위큐 최소값과 section 값 비교
        int count = 1;
        for(int zone:section){
            if( (zone - paintWall.peek()) >= m){
                count += 1;

                //queue reset 
                paintWall.clear();
                for(int i = zone; i<zone +m; i++){
                    paintWall.add(i);
                }
            }
        return count;
    }
    //Test
    public static void main(String[] args) {
        Painting painting = new Painting();
        System.out.println(painting.solution(8, 4, new int[]{2,3,6,7,8}));
        System.out.println(painting.solution(5, 4, new int[]{1,3}));
        System.out.println(painting.solution(4, 1, new int[]{1, 2, 3, 4}));
        System.out.println(painting.solution(4, 4, new int[]{1, 2, 3, 4}));
        System.out.println(painting.solution(2, 1, new int[]{1}));
    }
}
