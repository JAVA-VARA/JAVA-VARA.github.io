## 03. 회귀 알고리즘과 모델 규제

## **03–1. K-최근접 이웃 회귀**

지도 학습 알고리즘은 **분류 / 회귀**로 나뉜다. 그 중 **회귀는 정해진 클래스가 없고, 임의의 수치를 출력**하며, 실생활에 사용 되는 예로는 경제 성장률, 배달 도착 시간 예측 등이 있다.

```
import numpy as np

perch_length = np.array([8.4, 13.7, 15.0, 16.2, 17.4, 18.0, 18.7, 19.0, 19.6, 20.0, 21.0,
       21.0, 21.0, 21.3, 22.0, 22.0, 22.0, 22.0, 22.0, 22.5, 22.5, 22.7,
       23.0, 23.5, 24.0, 24.0, 24.6, 25.0, 25.6, 26.5, 27.3, 27.5, 27.5,
       27.5, 28.0, 28.7, 30.0, 32.8, 34.5, 35.0, 36.5, 36.0, 37.0, 37.0,
       39.0, 39.0, 39.0, 40.0, 40.0, 40.0, 40.0, 42.0, 43.0, 43.0, 43.5,
       44.0])
perch_weight = np.array([5.9, 32.0, 40.0, 51.5, 70.0, 100.0, 78.0, 80.0, 85.0, 85.0, 110.0,
       115.0, 125.0, 130.0, 120.0, 120.0, 130.0, 135.0, 110.0, 130.0,
       150.0, 145.0, 150.0, 170.0, 225.0, 145.0, 188.0, 180.0, 197.0,
       218.0, 300.0, 260.0, 265.0, 250.0, 250.0, 300.0, 320.0, 514.0,
       556.0, 840.0, 685.0, 700.0, 700.0, 690.0, 900.0, 650.0, 820.0,
       850.0, 900.0, 1015.0, 820.0, 1100.0, 1000.0, 1100.0, 1000.0,
       1000.0])

import matplotlib.pyplot as plt
plt.scatter(perch_length, perch_weight)
plt.xlabel('length')
plt.ylabel('weight')
plt.show()

from sklearn.model_selection import train_test_split

train_input, test_input, train_target, test_target = train_test_split(perch_length, perch_weight, random_state=42)

train_input = train_input.reshape(-1,1) #-1 : 배열 크기 자동 지정(원소 개수로 모두 채우라는 의미미)
test_input = test_input.reshape(-1,1)
print(train_input.shape, test_input.shape)

from sklearn.neighbors import KNeighborsRegressor

knr = KNeighborsRegressor()
#k 최근접 이웃 회귀 모델을 훈련합니다.

knr.fit(train_input, train_target)
print(knr.score(test_input, test_target))

from sklearn.metrics import mean_absolute_error

#테스트 세트에 대한 예측을 만든다
test_prediction= knr.predict(test_input)
#테스트 세트에 대한 평균 절댓값 오차 계산
mae = mean_absolute_error(test_target, test_prediction)
print(mae)

#결과에서 예측이 평균적으로 19g 정도 타깃값과 다르다.

print(knr.score(train_input, train_target))

knr.n_neighbors =3 #k값 3으로 설정

knr.fit(train_input, train_target)
print(knr.score(train_input, train_target))
print(knr.score(test_input, test_target))

#예제
knr = KNeighborsRegressor()
#5에서 45까지 x 좌표를 만든다/
x=np.arange(5,45).reshape(-1,1)

for n in [1,5,10]:
  knr.n_neighbors = n
  knr.fit(train_input, train_target)

  prediction = knr.predict(x)

  plt.scatter(train_input, train_target)
  plt.plot(x,prediction)
  plt.title('n_neighbors = {}'.format(n))
  plt.xlabel('length')
  plt.ylabel('weight')
  plt.show()
```

## **과대 적합: score train set >>>> test set**

훈련에만 잘 맞는 모델, 실전 투입하여 새로운 예측을 만들 때 잘 작동 하지 않는다.

solution) 모델을 “덜” 복잡하게 만든다.(k 값을 높인다.)

## 과소 적합:score train set <<<< test set

모델이 너무 단순하여 훈련이 적절히 되지 않은 경우

solution)

- 모델을 복잡하게 만든다.(훈련 세트에 더 잘 맞게 만들면, test set 점수 낮아진다.)
- 복잡하게 만드는 방법: 이웃의 개수 k 줄인다. k 수 줄이면 국지적인 패턴에 민감해지고, k 증가할수록 데이터 전반에 있는 일반적인 패턴을 따른다.
- k기본값이 5이므로, 3으로 낮춘다.

reshape : 배열의 크기를 바꾸는 메소드. 바꾸고자 하는 배열의 크기를 매개변수로 전달. 바꾸기 전/후의 배열 원소 개수는 동일해야한다.

## 03–2. 선형 회귀

```
import numpy as np

perch_length = np.array([8.4, 13.7, 15.0, 16.2, 17.4, 18.0, 18.7, 19.0, 19.6, 20.0, 21.0,
       21.0, 21.0, 21.3, 22.0, 22.0, 22.0, 22.0, 22.0, 22.5, 22.5, 22.7,
       23.0, 23.5, 24.0, 24.0, 24.6, 25.0, 25.6, 26.5, 27.3, 27.5, 27.5,
       27.5, 28.0, 28.7, 30.0, 32.8, 34.5, 35.0, 36.5, 36.0, 37.0, 37.0,
       39.0, 39.0, 39.0, 40.0, 40.0, 40.0, 40.0, 42.0, 43.0, 43.0, 43.5,
       44.0])
perch_weight = np.array([5.9, 32.0, 40.0, 51.5, 70.0, 100.0, 78.0, 80.0, 85.0, 85.0, 110.0,
       115.0, 125.0, 130.0, 120.0, 120.0, 130.0, 135.0, 110.0, 130.0,
       150.0, 145.0, 150.0, 170.0, 225.0, 145.0, 188.0, 180.0, 197.0,
       218.0, 300.0, 260.0, 265.0, 250.0, 250.0, 300.0, 320.0, 514.0,
       556.0, 840.0, 685.0, 700.0, 700.0, 690.0, 900.0, 650.0, 820.0,
       850.0, 900.0, 1015.0, 820.0, 1100.0, 1000.0, 1100.0, 1000.0,
       1000.0])

from sklearn.model_selection import train_test_split

train_input, test_input, train_target, test_target = train_test_split(
    perch_length, perch_weight, random_state=42
)

#train set, test set 을 2차원 배열로 바꾼다.
train_input = train_input.reshape(-1,1)
test_input = test_input.reshape(-1,1)

from sklearn.neighbors import KNeighborsRegressor
knr = KNeighborsRegressor(n_neighbors=3)

knr.fit(train_input, train_target)
print(knr.predict([[50]]))

import matplotlib.pyplot as plt

distances, indexes = knr.kneighbors([[50]])
plt.scatter(train_input, train_target)

plt.scatter(train_input[indexes], train_target[indexes], marker='D')

plt.scatter(50,1033,marker='^')
plt.xlabel('length')
plt.ylabel('weight')
plt.show()

print(np.mean(train_target[indexes]))

#샘플이 훈련 세트의 범위를 벗어나면 엉뚱한 값을 예측할 수 있다.

from sklearn.linear_model import LinearRegression
lr = LinearRegression()

lr.fit(train_input, train_target)

print(lr.predict([[50]]))

print(lr.coef_)
print(lr.intercept_)
#coef_ , intercept_를 모델 파라미터라고 부른다.
#최적의 모델 파라미터를 찾는 것이 머신러닝 알고리즘의 훈련 목표 => 모델 기반 학습
#k-최근접 이웃은 모델 파라미터 없음. 사례 기반 학습.

plt.scatter(train_input, train_target)
plt.plot([15.50],)

plt.plot([15,50], [15*lr.coef_+lr.intercept_, 50*lr.coef_+lr.intercept_])
plt.scatter(50, 1241.8, marker='^')
plt.xlabel('length')
plt.ylabel('weight')
plt.show()

#다항회귀: 다항식을 사용한 선형 회귀귀
train_poly = np.column_stack((train_input**2, train_input))
test_poly = np.column_stack((test_input**2, test_input))

print(train_poly.shape, test_poly.shape)

lr = LinearRegression()
lr.fit(train_poly, train_target)
print(lr.predict([[50**2, 50]]))

print(lr.coef_, lr.intercept_)

point = np.arange(15,50)
plt.scatter(train_input, train_target)
plt.plot(point, 1.01*point**2 - 21.6*point+116.05)

plt.scatter(50, 1574, marker = '^')
plt.xlabel('length')
plt.ylabel('weight')
plt.show()

print(lr.score(train_poly, train_target))
print(lr.score(test_poly, test_target))
```
