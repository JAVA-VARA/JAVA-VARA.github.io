import java.util.*;

public class CompressedMsg {
    public int[] solution(String msg) {
        //색인 만들기
        Map<String, Integer> dic = new HashMap<>();
        String basic = "ABCDEFGHIJKLMNOPQRSTUVWXYZ";
        for (int i = 1; i <= basic.length(); i++) {
            dic.put(basic.substring(i - 1, i), i);
        }

        int idx = 27;
        ArrayList<Integer> arrayList = new ArrayList<>();
        int i = 0;
        while (i < msg.length()) {
            int j = i + 1;
            while (j <= msg.length()) {

                String checkMsg = msg.substring(i, j);

                if (!dic.containsKey(checkMsg)) {
                    dic.put(checkMsg, idx++);
                    String previousMsg = checkMsg.substring(0, checkMsg.length() - 1);
                    arrayList.add(dic.get(previousMsg));
                    break;
                }

                if (j == msg.length()) {
                    arrayList.add(dic.get(checkMsg));
                }
                j++;
            }
            i = j - 1;
        }


        int[] answer = new int[arrayList.size()];
        int k = 0;
        for(int index : arrayList){
            answer[k] = index;
            k++;
        }
        return answer;
    }

    public static void main(String[] args) {
        CompressedMsg compression = new CompressedMsg();
        System.out.println(Arrays.toString(compression.solution("KAKAO")));
        System.out.println(Arrays.toString(compression.solution("TOBEORNOTTOBEORTOBEORNOT")));
        System.out.println(Arrays.toString(compression.solution("ABABABABABABABAB")));
    }
}
