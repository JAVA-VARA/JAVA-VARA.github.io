class Node:

    def __init__(self, item):
        self.data = item
        self.next = None


class LinkedList:

    def __init__(self):
        self.nodeCount = 0
        self.head = Node(None)
        self.tail = None
        self.head.next = self.tail


    def traverse(self):
        result = []
        curr = self.head
        while curr.next:
            curr = curr.next
            result.append(curr.data)
        return result


    def getAt(self, pos):
        if pos < 0 or pos > self.nodeCount:
            return None

        i = 0
        curr = self.head
        while i < pos:
            curr = curr.next
            i += 1

        return curr


    def insertAfter(self, prev, newNode):
        newNode.next = prev.next
        if prev.next is None:
            self.tail = newNode
        prev.next = newNode
        self.nodeCount += 1
        return True


    def insertAt(self, pos, newNode):
        if pos < 1 or pos > self.nodeCount + 1:
            return False

        if pos != 1 and pos == self.nodeCount + 1:
            prev = self.tail
        else:
            prev = self.getAt(pos - 1)
        return self.insertAfter(prev, newNode)


   def popAfter(self, prev):
       #prev 의 next가 가리키는 것이 없다면,
       #삭제할 node가 없는 것이므로 none 을 return한다.
        if prev.next == None:
            return None
        
        #prev.next가 가리키는 것을 curr이라고 한다.
        curr = prev.next
        
        #prev.next가 가리키고 있는 것을 curr.next로 옮긴다.
        prev.next = curr.next
        
        if prev.next == None:
            #prev.next가 가리키는 것이 없다면,
            #삭제된 node가 마지막 node였으므로
            self.tail = prev

        return curr.data
        #삭제한 node의 data를 반환한다.

            raise IndexError
        #pos가 범위를 벗어나면 indexError를 반환한다.
         
        return self.popAfter(self.getAt(pos-1))

class Node:

    def __init__(self, item):
        self.data = item
        self.prev = None
        self.next = None


class DoublyLinkedList:

    def __init__(self):
        self.nodeCount = 0
        self.head = Node(None)
        self.tail = Node(None)
        self.head.prev = None
        self.head.next = self.tail
        self.tail.prev = self.head
        self.tail.next = None


    def reverse(self):
        result = []
        #빈 리스트를 만든다.
        curr = self.tail
        #node의 tail이 가리키고 있는 곳을 curr이라고 한다.
        while curr.prev.prev:
        #curr.prev.prev가 없을 때까지, 즉 head node가 나올 때까지 반복
            curr=curr.prev 
            #curr은 이전 node를 가리키게하고
            result.append(curr.data)
            #해당 노드의 data를 result에 추가한다.
        return result
        #리스트를 역방향으로 순회하며 각 노드의 값을 확인할 수 있다.
       

    def getAt(self, pos):
        if pos < 0 or pos > self.nodeCount:
            return None

        if pos > self.nodeCount // 2:
            i = 0
            curr = self.tail
            while i < self.nodeCount - pos + 1:
                curr = curr.prev
                i += 1
        else:
            i = 0
            curr = self.head
            while i < pos:
                curr = curr.next
                i += 1

        return curr


    def insertAfter(self, prev, newNode):
    #prev 라는 노드 다음에 node 삽입
        next = prev.next
        #prev 노드의 다음 노드를 next 라고함.
        newNode.prev = prev
        #새로 삽입하는 newNode가 가리키는 prev를 prev라고함.
        newNode.next = next
        #새로 삽입하는 newNode가 가리키는 next를 next라고함.
        prev.next = newNode
        #prev의 next는 newNode 임
        next.prev = newNode
        #next의 이전 노드는 newNode임.
        self.nodeCount += 1
        #추가했으므로 nodeCount +1
        return True


    def insertAt(self, pos, newNode):
        if pos < 1 or pos > self.nodeCount + 1:
            return False

        prev = self.getAt(pos - 1)
        return self.insertAfter(prev, newNode)

(Q)popAfter(), popBefore() 메소드 사용 시 끊어지게 되는 node의 값을 none으로 설정 시 error가 발생하여 의문점을 가지고 있었는데, QnA 게시판에서 의문점에 대한 답변을 찾아 기록한다.

(A)삭제되는 노드를 가리키고 있던 로컬 변수 (local variable) `curr` 에 굳이 `None` 을 대입해야 할 이유는 없습니다. 코드의 실행이 로컬 변수의 유효 범위를 벗어나게 되면 이 참조 (reference) 는 끊어지고 해당 객체는 더 이상 메모리에 남아 있지 않게 된다고 간주하면 됩니다.

class ArrayStack:

    def __init__(self):
        self.data = []

    def size(self):
        return len(self.data)

    def isEmpty(self):
        return self.size() == 0

    def push(self, item):
        self.data.append(item)

    def pop(self):
        return self.data.pop()

    def peek(self):
        return self.data[-1]


def solution(expr):
    match = {
        ')': '(',
        '}': '{',
        ']': '['
    }

    S = ArrayStack()
    for c in expr:
        if c in '({[':
            S.push(c)

        elif c in match:
            if S.size()==0:
                return False
            else:
                t = S.pop()
                if t != match[c]:
                    return False

      return S.isEmpty()
