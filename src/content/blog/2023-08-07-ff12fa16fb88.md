package Programmers_Solution;

import java.util.LinkedList;
import java.util.Queue;
import java.util.*;

public class Fruit {
    public int solution(int k, int m, int[] score) {

        //내림 차순으로 score[] 정렬
        Integer[] arr = Arrays.stream(score).boxed().toArray(Integer[]::new);
        Arrays.sort(arr, Collections.reverseOrder());

        //정렬된 score[]를 Queue 에 삽입
        Queue<Integer> queue = new LinkedList<>(Arrays.asList(arr));

        List<Integer> furit_list = new ArrayList<>();
        int answer=0;
        //한 상자에 넣어야 하는 과일 수 보다 남은 과일 수가 적으면 0 반환
        if (score.length <= m) {
            return 0;
        }
        
        //과일 상자가 빌 떄까지 묶음 + 과일 상자 값 계산
        else while (!queue.isEmpty()) {

            for (int i = 0; i < m; i++) {
                furit_list.add(queue.poll());
            }

            answer += m * Collections.min(furit_list);
            furit_list.clear();

            //한 상자에 넣어야 하는 과일 수 보다 남은 과일 수가 적으면 break
            if (queue.size() < m) break;
        }

        return answer;
    }

    public static void main(String[] args) {
        Fruit fruit = new Fruit();
        System.out.println(fruit.solution(3, 4, new int[] {1,2,3,1,2,3,1}));
        System.out.println(fruit.solution(4, 3, new int[] {4, 1, 2, 2, 4, 4, 4, 4, 1, 2, 4, 2}));
        System.out.println(fruit.solution(3, 8, new int[] {1,2,3,1,2,3,1}));
    }
}

package Programmers_Solution;

import java.util.*;

public class Fail_rate {
    public int[] solution(int N, int[] stages) {

        //int[] to List로 변환
        Integer[] arr = Arrays.stream(stages).boxed().toArray(Integer[]::new);

        //각 층을 key값으로 설정
        Map<Integer, Integer> map = new HashMap<>();
        for(int i=1;i<=N+1;i++){
            map.put(i,0);
        }

        //각 층별 플레이어 숫자 반환(N+1 층까지)
        for(int i : arr){
            int count = map.get(i);
            map.put(i, count+1);
        }
//        System.out.println(map);

        //각 층별 실패율 반환
        Map<Integer, Double> failureRates = new HashMap<>();
        int totalPlayers = stages.length;
        for (int i = 1; i <= N; i++) {
            if(totalPlayers == 0){
                failureRates.put(i, 0.0);
            }else {
                int stagePlayers = map.get(i);
                double failureRate = (double) stagePlayers / totalPlayers;
                failureRates.put(i, failureRate);
                totalPlayers -= stagePlayers;
            }
        }
//        System.out.println(failureRates);

        List<Integer> result = new ArrayList<>(failureRates.keySet());
        result.sort(new Comparator<Integer>(){
            @Override
            public int compare(Integer o1, Integer o2){
                return failureRates.get(o2).compareTo(failureRates.get(o1));
            }
        });

        return result.stream()
                .mapToInt(Integer::intValue)
                .toArray();
    }
}
