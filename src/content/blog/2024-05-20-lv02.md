### Sol1

- 실패

```
import java.util.*;
public class Solution {
    public boolean solution(String[] phone_book) {
        boolean answer = true;
        //#1 번호 길이 확인
        Set<Integer> sizeOfPhoneNums = new HashSet<>();
        for(String phoneNumber : phone_book){
            int sizeOfPhoneNum = phoneNumber.length();
            sizeOfPhoneNums.add(sizeOfPhoneNum);
        }
        
        //#2 전화번호를 길이별로 잘라서 add
        LinkedList<String> list = new LinkedList<>();
        for(String phoneNumber : phone_book){
            for(int x : sizeOfPhoneNums){
                if(phoneNumber.length() < x){
                    continue;
                }
                String prefix = phoneNumber.substring(0,x);
                list.add(prefix);
            }
        }
        
        //#결과 확인
        Set<String> set = new HashSet<>(list);
        if(list.size() != set.size()){
            answer = false;
        }
        return answer;
    }
}
```

package Programmers_Solution.LV02;

import java.util.*;

public class PhoneNumList {
    public boolean solution(String[] phone_book) {
        boolean answer = true;

        //#1
        Map<String, Integer> map = new HashMap<>();
        for(String phoneNum : phone_book){
            map.put(phoneNum, 0);
        }

        //#2 번호 길이 확인
        Set<Integer> sizeOfPhoneNums = new HashSet<>();
        for(String phoneNumber : phone_book){
            int sizeOfPhoneNum = phoneNumber.length();
            sizeOfPhoneNums.add(sizeOfPhoneNum);
        }

        //#3 전화번호를 길이별로 잘라서 접두어 추출
        LinkedList<String> list = new LinkedList<>();
        for(String phoneNumber : phone_book){
            for(int x : sizeOfPhoneNums){
                if(phoneNumber.length() <= x){
                    continue;
                }
                String prefix = phoneNumber.substring(0,x);

                if(map.containsKey(prefix)){
                    return false;
                }
            }
        }
        return answer;
    }
}
