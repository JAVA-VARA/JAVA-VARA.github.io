### Solution1

```
import java.util.LinkedList;

public class NewsClustering {
    public int solution(String str1, String str2) {
        int answer = 0;
        
        LinkedList<String> list1 = new LinkedList<>();
        LinkedList<String> list2 = new LinkedList<>();

        String strL1 = str1.toLowerCase();
        String strL2 = str2.toLowerCase();

        //다중집합 생성
        for(int i =0; i < strL1.length()-1; i++){
            list1.add(strL1.substring(i,i+2));
        }
        //원소 중에 특수문자, 숫자, 공백 포함되어 있으면 제거
        for(int i =0;  i < list1.size(); i++){
            if(list1.get(i).matches(".*[0-9_\\s\\p{Punct}].*")){
                list1.remove(i);
                i--;
            }
        }

        for(int i =0; i < strL2.length()-1; i++){
            list2.add(strL2.substring(i,i+2));
        }
        for(int i =0;  i < list2.size(); i++){
            if(list2.get(i).matches(".*[0-9_\\s\\p{Punct}].*")){
                list2.remove(i);
                i--;
            }
        }
        
        //분자, 분모 모두 0인 경우
        if(list1.size() == 0 && list2.size() == 0){
            return 65536;
        }

        int denominator = list1.size() + list2.size();
        //그 외
          
        //교집합
        int numerator=0;
        for(String x : list1){
            for(String y : list2){
                if(x.equals(y)){
                    list2.remove(y);
                    numerator++;
                    break;
                }
            }
        }
        //합집합
        int  union = denominator - numerator;

        //결과
        answer = (int) Math.floor(((double) numerator / union) * 65536);
        return answer;
    }

    public static void main(String[] args) {
        NewsClustering newsClustering = new NewsClustering();
        System.out.println(newsClustering.solution("FRANCE", "french"));
        System.out.println(newsClustering.solution("handshake", "shake hands"));
        System.out.println(newsClustering.solution("aa1+aa2", "AAAA12"));
        System.out.println(newsClustering.solution("E=M*C^2", "e=m*c^2"));
    }
}

### Solution2

동일한 코드를 별도의 메서드(refinedStringList)로 작성

```
package Programmers_Solution.LV02;

import java.util.LinkedList;

public class NewsClustering {
    public int solution(String str1, String str2) {
        LinkedList<String> list1 = refinedStringList(str1);
        LinkedList<String> list2 = refinedStringList(str2);

        if(list1.size() == 0 && list2.size() == 0){
            return 65536;
        }

        int denominator = list1.size() + list2.size();

        //교집합
        int numerator=0;
        for(String x : list1){
            for(String y : list2){
                if(x.equals(y)){
                    list2.remove(y);
                    numerator++;
                    break;
                }
            }
        }
        //합집합
        int  union = denominator - numerator;

        //결과
        int answer = (int) Math.floor(((double) numerator / union) * 65536);
        return answer;
    }
    
    //다중 집합 생성 && 특수문자, 숫자 포함한 문자열 제거하는 메서
    private LinkedList refinedStringList(String title){
        LinkedList<String> list = new LinkedList<>();
        String strL1 = title.toLowerCase();
        for(int i =0; i < strL1.length()-1; i++){
            list.add(strL1.substring(i,i+2));
        }
        //원소 중에 특수문자, 숫자, 공백 포함되어 있으면 제거
        for(int i =0;  i < list.size(); i++){
            if(list.get(i).matches(".*[0-9_\\s\\p{Punct}].*")){
                list.remove(i);
                i--;
            }
        }
        return list;
    }
}
