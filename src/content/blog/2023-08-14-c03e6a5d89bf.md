[**코딩테스트 연습 - 대충 만든 자판**  
*휴대폰의 자판은 컴퓨터 키보드 자판과는 다르게 하나의 키에 여러 개의 문자가 할당될 수 있습니다. 키 하나에 여러 문자가 할당된 경우, 동일한 키를 연속해서 빠르게 누르면 할당된 순서대로 문자가 바뀝니다. 예를…*school.programmers.co.kr](https://school.programmers.co.kr/learn/courses/30/lessons/160586)[](https://school.programmers.co.kr/learn/courses/30/lessons/160586)

## Solution

```
import java.util.Arrays;
import java.util.HashMap;
import java.util.LinkedList;

public class Keyboard {
    public int[] solution(String[] keymap, String[] targets) {
        //원하는 알파벳에 접근하기 위한 최소 클릭 수
        HashMap<Character, Integer> keyboardMap = new HashMap<>();
        for (String str : keymap) {
            for (int i = 0; i < str.length(); i++) {
                char key = str.charAt(i);
                if (!keyboardMap.containsKey(key)) {
                    keyboardMap.put(key, i + 1);
                    continue;
                }
                if (keyboardMap.containsKey(key) && keyboardMap.get(key) > i + 1) {
                    keyboardMap.put(key, i + 1);
                }
            }
        }
        //문자별 touch가 필요한 횟수
        LinkedList<Integer> countTouch = new LinkedList<>();
        for (String str : targets) {
            int touch = 0;
            for (int i = 0; i < str.length(); i++) {
                if (keyboardMap.containsKey(str.charAt(i))) {
                    touch += keyboardMap.get(str.charAt(i));

                } else {
                    touch = -1;
                    break;
                }
            }
            countTouch.add(touch);
        }
        return countTouch.stream().mapToInt(touch -> touch).toArray();
    }
    //test
    public static void main(String[] args) {
        Keyboard keyboard = new Keyboard();
        System.out.println(Arrays.toString(keyboard.solution(new String[]{"ABACD", "BCEFD"}, new String[]{"ABCD", "AABB"}))); //[9, 4]
        System.out.println(Arrays.toString(keyboard.solution(new String[]{"AA"}, new String[]{"AA", "BAA"}))); //[-1]
        System.out.println(Arrays.toString(keyboard.solution(new String[]{"AGZ", "BSSS"}, new String[]{"ASA", "BGZ"}))); //[4, 6]
    }
}
```
