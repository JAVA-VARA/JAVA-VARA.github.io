### Solution1

```
import java.util.LinkedList;
import java.util.Stack;

public class rotate {
    public int solution(String s) {
        int answer = 0;
        rotate rotate = new rotate();

        char[] charArray = s.toCharArray();
        LinkedList<Character> characterLinkedList  = new LinkedList<>();
        for(char c : charArray){
            characterLinkedList .add(c);
        }

        for(int i=0; i<characterLinkedList.size(); i++){
            //올바른 모양인지 check
            if(rotate.checkRightBracket(characterLinkedList)){
                answer ++;
            }
            
            //배열 rotation
            characterLinkedList.add(characterLinkedList.get(0));
            characterLinkedList.remove(0);
        }

        return answer;
    }

    public static void main(String[] args) {
        rotate rotate = new rotate();
        System.out.println(rotate.solution("[](){}"));
        System.out.println(rotate.solution("}]()[{"));
        System.out.println(rotate.solution("[)(]"));
        System.out.println(rotate.solution("}}}"));
    }

    private boolean checkRightBracket(LinkedList<Character> s){
        Stack<Character> stack = new Stack<>();

        for(char c : s){
            if(stack.isEmpty()){
                stack.push(c);
                continue;
            }
            if( (stack.peek() == '[' && c ==']') | (stack.peek() == '{' && c =='}') | (stack.peek() == '(' && c ==')') ){
                stack.pop();
            }else {
                stack.push(c);
            }
        }
        return stack.isEmpty();
    }
}
```

### Solution2

isMatching 메서드를 추가하여 불필요하게 반복되는 부분을 삭제해주고 가독성을 높였다.

```
import java.util.LinkedList;
import java.util.Stack;

public class rotate {
    // 생략
    private boolean checkRightBracket(LinkedList<Character> bracketList){
        Stack<Character> stack = new Stack<>();
        for(char bracket : bracketList){
            if(!stack.isEmpty() && isMatching(stack.peek(), bracket)){
                stack.pop();
            }else {
                stack.push(bracket);
            }
        }
        return stack.isEmpty();
    }

    private boolean isMatching(char left, char right){
        return  (left == '[' && right ==']') |
                (left == '{' && right =='}') |
                (left == '(' && right ==')');
    }
}
```
