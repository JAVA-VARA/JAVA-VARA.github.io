## [Sol 1]

```
package Programmers_Solution.LV01;

import java.util.ArrayList;
import java.util.Arrays;
import java.util.Comparator;

public class DataAnalaysis {
    public int[][] solution(int[][] data, String ext, int val_ext, String sort_by) {

        int idx = switch (ext) {
            case "code" -> 0;
            case "date" -> 1;
            case "maximum" -> 2;
            case "remain" -> 3;
            default -> -1;
        };

        ArrayList<int[]> result = new ArrayList<>();
        for (int[] datum : data) {
            if (datum[idx] < val_ext) {
                result.add(datum);
            }
        }

        int sortedByIdx = switch (sort_by) {
            case "code" -> 0;
            case "date" -> 1;
            case "maximum" -> 2;
            case "remain" -> 3;
            default -> -1;
        };

        result.sort(Comparator.comparingInt(arr -> arr[sortedByIdx]));

        // 결과
        int i = 0;
        int[][] answer = new int[result.size()][];
        for (int[] array : result) {
            answer[i] = array;
            i++;
        }

        return answer;
    }

    public static void main(String[] args) {
        DataAnalaysis dataAnalaysis = new DataAnalaysis();
        System.out.println(Arrays.deepToString(
                dataAnalaysis.solution(
                        new int[][]{{1, 20300104, 100, 80}, {2, 20300804, 847, 37}, {3, 20300401, 10, 8}},
                        "date",
                        20300501,
                        "remain")));
    }
}
```
