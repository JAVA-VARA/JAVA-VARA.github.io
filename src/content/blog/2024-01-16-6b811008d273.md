public class BandageGame {

    public int solution(int[] bandage, int health, int[][] attacks) {

        //총 게임 시간
        int lastAttack = attacks.length-1;
        int totalTime = attacks[lastAttack][0];

        //초기값 셋팅
        int duration = 0;
        int currentHp = health;
        int attackTime = 0;

        for(int i = 0; i <= totalTime; i++){
            System.out.println("Time: " + i);

            //공격 안받을 때 체력 증가
            if(i != attacks[attackTime][0]){
                //기본 회복
                currentHp += bandage[1];
                duration += 1;
                //보너스 회복
                if(duration == bandage[0]){
                    currentHp += bandage[2];
                    duration = 0;
                }
                //최대 hp 초과 못하도록 설정
                if(currentHp >= health){
                    currentHp = health;
                }
                System.out.println("currentHp: " + currentHp);
            }

            //공격 받을 때 체력 감소
            else{
                currentHp -= attacks[attackTime][1];
                System.out.println("attackTime: " + attackTime);
                System.out.println("currentHp: " + currentHp);
                attackTime += 1;
                duration = 0;
            }

            if(currentHp <= 0){
                return -1;
            }
        }

        return currentHp;
    }

    public static void main(String[] args) {
        BandageGame bandageGame = new BandageGame();
        System.out.println(bandageGame.solution(
                new int[]{5, 1, 5},
                30,
                new int[][]{{2, 10}, {9, 15}, {10, 5}, {11, 5}}));

        System.out.println(bandageGame.solution(
                new int[]{3, 2, 7},
                20,
                new int[][]{{1, 15}, {5, 16}, {8, 6}}));

        System.out.println(bandageGame.solution(
                new int[]{4, 2, 7},
                20,
                new int[][]{{1, 15}, {5, 16}, {8, 6}}));

        System.out.println(bandageGame.solution(
                new int[]{1, 1, 1},
                5,
                new int[][]{{1, 2}, {3, 2}}));
    }
}
