### Solution 1

```
import java.util.Arrays;
public class binaryConversion {
    public int[] solution(String s) {
        int[] answer = new int[2];
        // 주어진 문자열이 "1"이 될 때까지 반복
        while (!s.equals("1")) {
            // 0 갯수 카운트
            for(char c : s.toCharArray()){
                if(c == '0'){
                    answer[1] +=1;
                }
            }

            // 0 제거 및 길이를 이진수로 변환
            s = s.replaceAll("0","");
            int length = s.length();
            s = Integer.toBinaryString(length);
            answer[0]++;
        }
        return answer;
    }

    public static void main(String[] args) {
        binaryConversion bc = new binaryConversion();
        System.out.println(Arrays.toString(bc.solution("0111010")));
        System.out.println(Arrays.toString(bc.solution("110010101001"))); //[3,8]
        System.out.println(Arrays.toString(bc.solution("01110"))); //[3,3]
        System.out.println(Arrays.toString(bc.solution("1111111"))); //[4,1]
    }

}
```
