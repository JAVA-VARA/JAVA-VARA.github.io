```
package Programmers_Solution.Skilltest;

public class SkillTest3 {
    public double solution(int[] arr) {
        double answer = 0;

        int result = 0;
        for(int num : arr){
            result += num;
        }

        answer = (double) result / arr.length;

        return answer;
    }

    public static void main(String[] args) {
        SkillTest3 skilltest3 = new SkillTest3();
        System.out.println(skilltest3.solution(new int[] {1,2,3,4})); //2.5
        System.out.println(skilltest3.solution(new int[] {5,5})); //5
    }

}
```

```
package Programmers_Solution.Skilltest;

import java.util.Arrays;
import java.util.LinkedList;
import java.util.Queue;

public class SkillTest4 {
    public String solution(String[] cards1, String[] cards2, String[] goal) {
        String answer = "";

        Queue<String> deck1 = new LinkedList<>(Arrays.asList(cards1));
        Queue<String> deck2 = new LinkedList<>(Arrays.asList(cards2));

        for(String word: goal){
            if(!deck1.isEmpty() && deck1.peek().equals(word)){
                deck1.poll();
                continue;

            }if(!deck2.isEmpty() && deck2.peek().equals(word)){
                deck2.poll();

            }else{
                return "No";
            }
        }
        answer = "Yes";

        return answer;
    }

    public static void main(String[] args) {
        SkillTest4 skilltest4 = new SkillTest4();
        System.out.println(skilltest4.solution(
                new String[] {"i", "drink", "water"},
                new String[] {"want", "to"},
                new String[] {"i", "want", "to", "drink"}));  //yes

        System.out.println(skilltest4.solution(
                new String[] {"i", "water", "drink"},
                new String[] {"want", "to"},
                new String[] {"i", "want", "to", "drink", "water"})); //no

        System.out.println(skilltest4.solution(
                new String[] {"to"},
                new String[] {"to"},
                new String[] {"to", "to"}));  //yes
    }
}
```

```
package Programmers_Solution.Skilltest;

public class SkillTest5 {
    boolean solution(String s) {
        boolean answer;

        char[] chars = s.toCharArray();

        int countP =0;
        int countY =0;
        for(char c : chars){
            if(c == 'p' || c == 'P'){
                countP += 1;
            }
            if(c == 'y' || c == 'Y'){
                countY += 1;
            }
        }
        answer = countP == countY || countP + countY == 0;


        return answer;
    }


    public static void main(String[] args) {
        SkillTest5 skillTest5 = new SkillTest5();
        System.out.println(skillTest5.solution("pPoooyY"));
        System.out.println(skillTest5.solution("Pyy"));
    }
}
```

```
package Programmers_Solution.Skilltest;

import java.util.Arrays;
import java.util.LinkedList;
import java.util.stream.IntStream;

public class SkillTest6 {
    public int[] solution(int[] arr, int divisor) {
        LinkedList<Integer> result = new LinkedList<>();
        for(int num:arr){
            if(num%divisor == 0){
                result.add(num);
            }
        }

        if(result.isEmpty()){
            return new int[] {-1};
        }

        //#1
        int[] answer = new int[result.size()];
        for(int i = 0; i <result.size(); i++){
            answer[i] = result.get(i);
        }
        //#2
        IntStream.range(0, result.size()).forEach(i -> answer[i] = result.get(i));

        Arrays.sort(answer);

        return answer;
    }
    public static void main(String[] args) {
        SkillTest6 skillTest6 = new SkillTest6();
        System.out.println(Arrays.toString(skillTest6.solution(new int[]{5, 9, 7, 10}, 5)));
        System.out.println(Arrays.toString(skillTest6.solution(new int[]{2, 36, 1, 3}, 1)));
        System.out.println(Arrays.toString(skillTest6.solution(new int[]{3,2,6}, 10)));
    }
}
```
