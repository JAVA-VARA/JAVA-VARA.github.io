### Java LV02

[**코딩테스트 연습 - 주차 요금 계산**  
*문제 설명 주차장의 요금표와 차량이 들어오고(입차) 나간(출차) 기록이 주어졌을 때, 차량별로 주차 요금을 계산하려고 합니다. 아래는 하나의 예시를 나타냅니다. 어떤 차량이 입차된 후에 출차된 내역이 없다면…*school.programmers.co.kr](https://school.programmers.co.kr/learn/courses/30/lessons/92341)[](https://school.programmers.co.kr/learn/courses/30/lessons/92341)

package Programmers_Solution.LV02;

import java.util.*;

public class ParkingFee {
    public int[] solution(int[] fees, String[] records) {
        List<String[]> list = new ArrayList<>();
        for (String record : records) {
            String[] input = record.split(" ");
            list.add(input);
        }

        Map<String, String[]> mapIn = new HashMap<>();
        Map<String, Integer> mapTotalUsage = new HashMap<>();
        for (String[] parkingInfo : list) {
            String[] time = parkingInfo[0].split(":");
            String carNumber = parkingInfo[1];
            String inAndOut = parkingInfo[2];

            //입차한 차량 등록
            if (inAndOut.equals("IN")) {
                mapIn.put(carNumber, time);
                continue;
            }
            //출차 차량 주차 시간
            int inHour = Integer.parseInt(mapIn.get(carNumber)[0]);
            int outHour = Integer.parseInt(time[0]);

            int inMin = Integer.parseInt(mapIn.get(carNumber)[1]);
            int outMin = Integer.parseInt(time[1]);

            int total;

            if (outMin >= inMin) {
                total = (outHour - inHour) * 60 + outMin - inMin;
            } else {
                total = (outHour - inHour - 1) * 60 + outMin - inMin + 60;
            }
            mapTotalUsage.put(carNumber, mapTotalUsage.getOrDefault(carNumber, 0) + total);

            mapIn.remove(carNumber);
        }
        //출차하지 않은 차량의 주차 시간
        for(String stay: mapIn.keySet()){
            int inHour = Integer.parseInt(mapIn.get(stay)[0]);
            int outHour = 23;

            int inMin = Integer.parseInt(mapIn.get(stay)[1]);
            int outMin = 59;

            int total = (outHour - inHour) * 60 + outMin - inMin;

            mapTotalUsage.put(stay, mapTotalUsage.getOrDefault(stay, 0) + total);

        }

        //차량별 요금 계산
        for(String key : mapTotalUsage.keySet()){
            int defaultTime  = fees[0];
            int basicFee = fees[1];
            if(mapTotalUsage.get(key) <= fees[0]){
                mapTotalUsage.put(key, basicFee);
            }else {
                int extraUsage = mapTotalUsage.get(key) - defaultTime;
                int extraFee = (int) Math.ceil( (double) extraUsage/fees[2])  * fees[3];
                int totalFee = basicFee + extraFee;
                mapTotalUsage.put(key, totalFee);
            }
        }

        //정렬
        int[] answer = new int[mapTotalUsage.size()];
        Set<String> keySet = mapTotalUsage.keySet();
        ArrayList<String> keyList = new ArrayList<>(keySet);
        Collections.sort(keyList);

        int i =0;
        for(String key: keyList){
            answer[i] = mapTotalUsage.get(key);
            i++;
        }
        System.out.println(Arrays.toString(answer));
        return answer;
    }
    public static void main(String[] args) {
        ParkingFee parkingFee = new ParkingFee();
        System.out.println(parkingFee.solution(
                new int[]{180, 5000, 10, 600},
                new String[]{"05:34 5961 IN", "06:00 0000 IN", "06:34 0000 OUT", "07:59 5961 OUT", "07:59 0148 IN", "18:59 0000 IN", "19:09 0148 OUT", "22:59 5961 IN", "23:00 5961 OUT"}));

        System.out.println(parkingFee.solution(
                new int[]{120, 0, 60, 591},
                new String[]{"16:00 3961 IN","16:00 0202 IN","18:00 3961 OUT","18:00 0202 OUT","23:58 3961 IN"}));

        System.out.println(parkingFee.solution(
                new int[]{1, 461, 1, 10},
                new String[]{"00:00 1234 IN"}));
    }
}

package Programmers_Solution.LV02;

import java.util.*;

public class Parking {
    public int[] solution(int[] fees, String[] records) {
        List<String[]> list = arrangedRecords(records);
        //입차 기록 저장할 map 생성
        Map<String, String[]> mapIn = new HashMap<>();
        //차량 별 이용시간, 금액 저장할 map 생성
        Map<String, Integer> mapTotalUsage = new HashMap<>();

        for (String[] parkingInfo : list) {
            String[] time = parkingInfo[0].split(":");
            String carNumber = parkingInfo[1];
            String inAndOut = parkingInfo[2];

            //입차
            if (inAndOut.equals("IN")) {
                mapIn.put(carNumber, time);
                continue;
            }

            //출차
            int total = calculateParkingTime(mapIn.get(carNumber), time);
            mapTotalUsage.put(carNumber, mapTotalUsage.getOrDefault(carNumber, 0) + total);
            mapIn.remove(carNumber);
        }

        //출차하지 않은 차
        for (String notExited : mapIn.keySet()) {
            int total = calculateParkingTime(mapIn.get(notExited), new String[]{"23", "59"});
            mapTotalUsage.put(notExited, mapTotalUsage.getOrDefault(notExited, 0) + total);
        }

        //차량별 이용 금액 최종 계산
        for (String key : mapTotalUsage.keySet()) {
            int totalFee = calculateParkingFee(mapTotalUsage.get(key), fees);
            mapTotalUsage.put(key, totalFee);
        }

        //정렬 후 반환
        int[] answer = new int[mapTotalUsage.size()];
        Set<String> keySet = mapTotalUsage.keySet();
        ArrayList<String> keyList = new ArrayList<>(keySet);
        Collections.sort(keyList);
        int index = 0;
        for (String key : keyList) {
            answer[index++] = mapTotalUsage.get(key);
        }
        System.out.println(Arrays.toString(answer));
        return answer;
    }
    // 차량 입/출차 기록을 리스트에 저장
    private List<String[]> arrangedRecords(String[] records) {
        List<String[]> list = new ArrayList<>();
        for (String record : records) {
            String[] input = record.split(" ");
            list.add(input);
        }
        return list;
    }

    // 차량의 주차 시간을 계산
    private int calculateParkingTime(String[] inTime, String[] outTime) {
        int inHour = Integer.parseInt(inTime[0]);
        int outHour = Integer.parseInt(outTime[0]);

        int inMin = Integer.parseInt(inTime[1]);
        int outMin = Integer.parseInt(outTime[1]);

        int total;
        if (outMin >= inMin) {
            total = (outHour - inHour) * 60 + outMin - inMin;
        } else {
            total = (outHour - inHour - 1) * 60 + outMin - inMin + 60;
        }
        return total;
    }

    // 주차 요금 계산
    private int calculateParkingFee(int totalUsage, int[] fees) {
        int defaultTime = fees[0];
        int basicFee = fees[1];
        if (totalUsage <= defaultTime) {
            return basicFee;

        } else {
            int extraUsage = totalUsage - defaultTime;
            int extraFee = (int) Math.ceil((double) extraUsage / fees[2]) * fees[3];
            return basicFee + extraFee;
        }
    }
}
