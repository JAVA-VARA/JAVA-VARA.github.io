               9.8, 10.5, 10.6, 11.0, 11.2, 11.3, 11.8, 11.8, 12.0, 12.2, 12.4, 13.0, 14.3, 15.0]

## 02–2. 데이터 전처리

```
from sklearn.metrics.pairwise import distance_metrics
fish_length = [25.4, 26.3, 26.5, 29.0, 29.0, 29.7, 29.7, 30.0, 30.0, 30.7, 31.0, 31.0, 
                31.5, 32.0, 32.0, 32.0, 33.0, 33.0, 33.5, 33.5, 34.0, 34.0, 34.5, 35.0, 
                35.0, 35.0, 35.0, 36.0, 36.0, 37.0, 38.5, 38.5, 39.5, 41.0, 41.0, 9.8, 
                10.5, 10.6, 11.0, 11.2, 11.3, 11.8, 11.8, 12.0, 12.2, 12.4, 13.0, 14.3, 15.0]
fish_weight = [242.0, 290.0, 340.0, 363.0, 430.0, 450.0, 500.0, 390.0, 450.0, 500.0, 475.0, 500.0, 
                500.0, 340.0, 600.0, 600.0, 700.0, 700.0, 610.0, 650.0, 575.0, 685.0, 620.0, 680.0, 
                700.0, 725.0, 720.0, 714.0, 850.0, 1000.0, 920.0, 955.0, 925.0, 975.0, 950.0, 6.7, 
                7.5, 7.0, 9.7, 9.8, 8.7, 10.0, 9.9, 9.8, 12.2, 13.4, 12.2, 19.7, 19.9]

import numpy as np

np.column_stack(([1,2,3], [4,5,6]))

fish_data = np.column_stack((fish_length, fish_weight))
fish_target = np.concatenate((np.ones(35),np.zeros(14)))
print(fish_data[0:5])
print(fish_target)

from sklearn.model_selection import train_test_split

train_input, test_input, train_target, test_target = train_test_split(fish_data, fish_target, random_state=42)

print(train_input.shape, test_input.shape)
print(train_target.shape, test_target.shape)

print(test_target)

train_input, test_input, train_target, test_target = train_test_split(fish_data, fish_target, stratify=fish_target, random_state=42)
#stratify 매개변수를 target data를 전달하면, 클래스 비율에 맞게 데이터를 나눠준다. 데이터가 작거나, 특정 클래스 샘플 개수가 적은 경우 유용.

print(test_target)

from sklearn.neighbors import KNeighborsClassifier
kn = KNeighborsClassifier()
kn.fit(train_input, train_target)
kn.score(test_input, test_target)

kn.predict([[25,150]])

import matplotlib.pyplot as plt
plt.scatter(train_input[:,0], train_input[:,1])
plt.scatter(25,150,marker='^')
plt.xlabel('length')
plt.ylabel('weight')
plt.show()

distance, indexes = kn.kneighbors([[25,150]])
#kneighbors(): 이웃까지의 거리와, 이웃 샘플의 인덱스를 나타내준다.

plt.scatter(train_input[:,0], train_input[:,1])
plt.scatter(25,150,marker='^')
plt.scatter(train_input[indexes,0], train_input[indexes, 1], marker = 'D')
plt.xlabel('length')
plt.ylabel('weight')
plt.show()


print(train_input[indexes])
print(train_target[indexes])
print(distance)

plt.scatter(train_input[:,0], train_input[:,1])
plt.scatter(25,150,marker='^')
plt.scatter(train_input[indexes,0], train_input[indexes, 1], marker = 'D')
plt.xlim((0,1000))
#x축의 간격을 y축과 동일하게 조정 => 산점도가 거의 일직선으로 나타난다.
#생선의 무게(y축)만 고려 대상이 된다.
#길이와 무게의 범위, 단위가 매우 다르다. 이를 두 특성의 "스케일"이 다르다고 한다.
plt.xlabel('length')
plt.ylabel('weight')
plt.show()

#알고리즘이 거리 기반인 경우, 기준이 다르면 올바르게 예측하기 어렵다.
#샘플 간 거리 영향을 많이 받으므로, 일정한 기준으로 맞춰줘야하며,
#이를 "데이터 전처리" 라고 한다.

#표준점수:각 특성값이 평균에서 표준편차의 몇 배만큼 떨어져 있는지 나타낸다.
#실제 특성값의 크기와 관계없이 동일한 조건으로 비교할 수 있다.

#분산: 데이터에서 평균을 뺀 값을 모두 제곱한 다음 평균을 내어 구함.
#표준편차는 분산의 제곱근으로, 데이터가 분산된 정도를 나타낸다.

mean = np.mean(train_input, axis=0) #평균 계산
std = np.std(train_input, axis=0) #표준편차 계산

#특성마다 값의 스케일이 다르므로 평군과 표준편차는 각 특성별로 계산해야 한다.
#axis = 0 으로 지정하면 행을 따라 각 열의 통계값을 계산한다.

print(mean,std)

train_scaled = (train_input - mean)/std

#전처리 데이터 모델 훈련
new = ([25, 150]-mean)/std
plt.scatter(train_scaled[:,0],train_scaled[:,1])
plt.scatter(new[0],new[1], marker='^')
plt.xlabel('length')
plt.ylabel('weight')
plt.show()

kn.fit(train_scaled, train_target)
test_scaled = (test_input-mean) / std

kn.score(test_scaled, test_target)

print(kn.predict([new]))

distances, indexes = kn.kneighbors([new])
plt.scatter(train_scaled[:,0], train_scaled[:,1])
plt.scatter(new[0], new[1], marker= '^')
plt.scatter(train_scaled[indexes,0], train_scaled[indexes,1], marker = 'D')
plt.xlabel('length')
plt.ylabel('weight')
plt.show()
```
