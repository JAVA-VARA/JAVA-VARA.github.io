package Programmers_Solution.LV02;

import java.util.Arrays;
import java.util.HashMap;
import java.util.Map;

public class LinkingEndWord2 {
    public int[] solution(int n, String[] words) {
        int[] answer = new int[2];

        Map<String, Integer> checkDuplicated = new HashMap<>();
        for(int i  = 0; i < words.length; i++){
            //중복 체크
            if(!checkDuplicated.containsKey(words[i])){
                checkDuplicated.put(words[i], i);
            }
            else {
                int stopIdx = i + 1;
                calculateResult(answer, n, stopIdx);
                break;
            }

            //끝말잇기 체크
            if(i == words.length - 1){
                break;
            }

            char endWith = words[i].charAt(words[i].length()-1);
            char startWith = words[i+1].charAt(0);
            if(endWith != startWith){

                int stopIdx = i + 2;
                calculateResult(answer, n, stopIdx);
                break;
            }
        }
        return answer;
    }

    private void calculateResult(int[] answer, int n, int stopIdx) {
        if (stopIdx % n == 0) {
            answer[0] = n;
            answer[1] = stopIdx / n;
        } else {
            answer[0] = stopIdx % n;
            answer[1] = stopIdx / n + 1;
        }
    }

    public static void main(String[] args) {
        LinkingEndWord2 linkingEndWord2 = new LinkingEndWord2();

        System.out.println(Arrays.toString(linkingEndWord2.solution(
                3,
                new String[]{
                        "tank", "kick", "know",
                        "wheel", "land", "dream",
                        "mother", "robot", "tank"})));
        System.out.println(Arrays.toString(linkingEndWord2.solution(
                5,
                new String[]{
                        "hello", "observe", "effect", "take", "either",
                        "recognize", "encourage", "ensure", "establish", "hang",
                        "gather", "refer", "reference", "estimate", "executive"})));
        System.out.println(Arrays.toString(linkingEndWord2.solution(
                2,
                new String[]{
                        "hello", "one",
                        "even", "never",
                        "now", "world",
                        "draw"})));

    }
}
