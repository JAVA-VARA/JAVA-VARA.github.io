public class Park {
    public long solution(int price, int money, int count) {
        long result;
        long total = 0;

        for (int i = 1; i <= count; i++) {
            total += price * i;
        }

        result = total - Long.valueOf(money);

        if (result >= 0) {
            result = result;
        } else {
            result = 0;
        }

        return result;
    }

import java.util.Arrays;
import java.util.Collections;

public class Budget {
    public int solution(int[] d, int budget) {
        
        // 필요한 예산의 총합.
        int total = 0;
        for (int i = 0; i < d.length; i++) {
            total += d[i];
        }

        // 필요한 예산 내림차순 정렬.
        d = Arrays.stream(d)
                .boxed().sorted(Collections.reverseOrder())
                .mapToInt(Integer::intValue).toArray();

        //예산 초과 시 가장 큰 비용부터 삭감.
        int count = 0; 
        for (int i = 0; i < d.length; i++) {
            if (total > budget) {
                total -= d[i];
                count += 1;
            } else {break;}
        }

        return d.length - count;
    }

    public static void main(String[] args) {
        Budget b = new Budget();
        System.out.println(b.solution(new int[] { 1, 3, 2, 5, 4 }, 9)); // 3
        System.out.println(b.solution(new int[] { 2, 2, 3, 3 }, 10)); // 4
    }

}
