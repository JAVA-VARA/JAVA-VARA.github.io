import java.util.ArrayList;
import java.util.LinkedList;
import java.util.Queue;

public class Solution {
    public int[] solution(int[] progresses, int[] speeds) {
        Queue<Integer> queue = new LinkedList<Integer>();
        ArrayList<Integer> al = new ArrayList<>();

        double a = 0;
        int b = 0;
        for (int i = 0; i < progresses.length; i++) {
            a = (100 - progresses[i]) / (double) speeds[i];
            b= (int) Math.ceil(a)
            queue.offer(b);
        }

        int temp = 0; 
        int count = 0;

        while (!queue.isEmpty()) {
            temp = queue.poll();
            count = 1;

            while ((!queue.isEmpty()) && (temp >= queue.peek())) {
                count += 1;
                queue.poll();
            }

            al.add(count);


        }

        int p = 0;
        int[] answer = new int[al.size()];

        for (int j : al) {
            answer[p] = j;
            p++;
        }
        return answer;
    }

    public static void main(String[] args) {
        Solution fd = new Solution();
        int[] sol1 = fd.solution(new int[] { 93, 30, 55 }, new int[] { 1, 30, 5 });
        int[] sol2 = fd.solution(new int[] { 95, 90, 99, 99, 80, 99 }, new int[] { 1, 1, 1, 1, 1, 1 });

        for (int value : sol1) {
            System.out.println(value);
        }

        System.out.println();

        for (int value : sol2) {
            System.out.println(value);
        }

    }

}
