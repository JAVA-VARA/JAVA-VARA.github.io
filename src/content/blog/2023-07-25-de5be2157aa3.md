package Programmers_Solution;

import java.util.*;

public class Reporting {
    public int[] solution(String[] id_list, String[] report, int k) {
        int[] answer = new int[id_list.length];

        //스페이스바 기준으로 나뉜 "신고한 user - troll"을 배열로 묶음.
        String[][] newReport = new String[report.length][2];
        for (int i = 0; i < report.length; i++) {
            newReport[i] = report[i].split(" ");
        }

        // user - troll을 key : value 값으로 묶음.
        Map<String, Set<String>> reportMap = new HashMap<>();
        for (String[] reportedUser : newReport) {
            String reporter = reportedUser[0];
            String troll = reportedUser[1];

            if (!reportMap.containsKey(reporter)) {
                reportMap.put(reporter, new HashSet<>());
            }
            reportMap.get(reporter).add(troll);
        }
        System.out.println(reportMap);

        // troll들 신고당한 횟수 count
        Map<String, Integer> trollCountMap = new HashMap<>();
        for (Set<String> trolls : reportMap.values()) {
            for (String troll : trolls) {
                trollCountMap.put(troll, trollCountMap.getOrDefault(troll, 0) + 1);
            }
        }

        //troll 목록 반환
        for (int i = 0; i < id_list.length; i++) {
            String user = id_list[i];
            if (reportMap.containsKey(user)) {
                Set<String> reportedTrolls = reportMap.get(user);

                //k번 이상 신고당한 경우에만 answer 에 1씩 더해 반환
                for (String reportedTroll : reportedTrolls) {
                    if (trollCountMap.containsKey(reportedTroll) && trollCountMap.get(reportedTroll) >= k) {
                        answer[i]++;
                    }
                }
            }
        }

        return answer;
    }

    public static void main(String[] args) {
        Reporting rr = new Reporting();

        System.out.println(Arrays.toString(rr.solution(
                new String[]{"muzi", "frodo", "apeach", "neo"},
                new String[]{"muzi frodo", "apeach frodo", "frodo neo", "muzi neo", "apeach muzi"},
                2)));

        System.out.println(Arrays.toString(rr.solution(
                new String[]{"con", "ryan"},
                new String[]{"ryan con", "ryan con", "ryan con", "ryan con"},
                3)));
    }
}

package Programmers_Solution;

import java.util.*;

public class Notification {
    public int[] solution(String[] id_list, String[] report, int k) {
        int[] answer = new int[id_list.length];

        Map<String,Integer> reported = new HashMap<>();  // 유저별 신고당한 횟수
        Map<String, ArrayList<String>> relation = new HashMap<>();   // 유저별 신고 관계

        for(int i=0;i<id_list.length;i++){  // 이용자를 위한 신고 서비스 생성
            relation.put(id_list[i],new ArrayList<>()); // 관계
            reported.put(id_list[i],0); // 횟수
        }

        //report 배열을 list로 변환(중복되는 신고 내용 삭제)
        List<String> reportList = new ArrayList<>(new HashSet<>(Arrays.asList(report)));
        //report라는 배열을 List로 변환-> 변환한 list를 HashSet에 넣어 중복된 내용 제거
        // -> 중복된 내용이 제거된 Set을 List에 삽입
        System.out.println(reportList);
        //report 배열에서 중복된 내용이 제건된 것이 List로 변환된다.

        for(int i = 0 ; i < reportList.size(); i++){
            String[] reporterAndTroll;
            reporterAndTroll = reportList.get(i).split(" ");
            System.out.println(Arrays.toString(reporterAndTroll));

            //신고한 사람의 목록을 만든다.
            ArrayList<String> reportingUser = relation.get(reporterAndTroll[0]);
            reportingUser.add(reporterAndTroll[1]);
            //신고한사람 : 신고당한사람 목록을 만든다.
            relation.put(reporterAndTroll[0], reportingUser);

            //신고당한 사람 횟수 count한다.
            reported.put(reporterAndTroll[1],reported.get(reporterAndTroll[1])+1);
        }
        System.out.println(relation);
        System.out.println(reported);

        //사용자가 신고한 troll을 가져온다.
        for(int i =0; i<id_list.length; i++){
            List<String> reportedUser = relation.get(id_list[i]);
            if(!reportedUser.isEmpty()){
                for(String user : reportedUser){
                    if(reported.get(user) >= k){
                        answer[i]++;
                    }
                }
            }

        }

        return answer;
    }

}
